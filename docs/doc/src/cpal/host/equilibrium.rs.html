3pub const DSD_EQUILIBRIUM_BYTE: u8 = 0x69;
4pub const U8_EQUILIBRIUM_BYTE: u8 = 0x80;
5
6/// Fill `buffer` with the equilibrium value for any `sample_format`.
7#[inline]
8pub fn fill_equilibrium(buffer: &mut [u8], sample_format: SampleFormat) {
9    macro_rules! fill_typed {
10        ($sample_type:ty) => {{
11            let sample_size = std::mem::size_of::<$sample_type>();
12
13            debug_assert_eq!(
14                buffer.len() % sample_size,
15                0,
16                "Buffer size must be aligned to sample size for format {:?}",
17                sample_format
18            );
19
20            let num_samples = buffer.len() / sample_size;
21            let equilibrium = <$sample_type as Sample>::EQUILIBRIUM;
22
23            // Safety: buffer length is verified to be a multiple of the sample size above.
24            let samples = unsafe {
25                std::slice::from_raw_parts_mut(
26                    buffer.as_mut_ptr() as *mut $sample_type,
27                    num_samples,
28                )
29            };
30
31            for sample in samples {
32                *sample = equilibrium;
33            }
34        }};
35    }
36
37    if sample_format.is_int() || sample_format.is_float() {
38        buffer.fill(0);
39    } else if sample_format == SampleFormat::U8 {
40        buffer.fill(U8_EQUILIBRIUM_BYTE);
41    } else if sample_format.is_dsd() {
42        buffer.fill(DSD_EQUILIBRIUM_BYTE);
43    } else {
44        // Multi-byte unsigned integer formats require a fill equal to the midpoint of their range.
45        debug_assert!(sample_format.is_uint());
46        match sample_format {
47            SampleFormat::U16 => fill_typed!(u16),
48            SampleFormat::U24 => fill_typed!(U24),
49            SampleFormat::U32 => fill_typed!(u32),
50            SampleFormat::U64 => fill_typed!(u64),
51            _ => unimplemented!(
52                "failed to fill equilibrium for unsupported unsigned format {sample_format:?}"
53            ),
54        }
55    }
56}
