1//! Error delivery helpers for audio callbacks.
2//!
3//! Pick the helper based on what you need:
4//! - Must not block (RT process callback): [`try_emit_error`]
5//! - Caller must not miss the error: [`emit_error`] (blocks until callback available)
6
7use std::sync::{Mutex, TryLockError};
8
9use crate::Error;
10
11/// Deliver an error, blocking until the callback is available.
12pub(crate) fn emit_error<E>(callback: &Mutex<E>, error: Error)
13where
14    E: FnMut(Error) + Send + ?Sized,
15{
16    let mut cb = callback.lock().unwrap_or_else(|e| e.into_inner());
17    (*cb)(error);
18}
19
20/// Try to deliver an error without blocking.
21///
22/// Returns `Ok(())` if the callback was invoked, or `Err(error)` if the lock was contended and
23/// the error could not be delivered.
24#[allow(dead_code)]
25pub(crate) fn try_emit_error<E>(callback: &Mutex<E>, error: Error) -> Result<(), Error>
26where
27    E: FnMut(Error) + Send + ?Sized,
28{
29    match callback.try_lock() {
30        Ok(mut cb) => {
31            (*cb)(error);
32            Ok(())
33        }
34        Err(TryLockError::Poisoned(e)) => {
35            (*e.into_inner())(error);
36            Ok(())
37        }
38        Err(TryLockError::WouldBlock) => Err(error),
39    }
40}
