20/// A [`Host`] provides access to the available audio devices on the system.
21///
22/// Each platform may have a number of available hosts depending on the system, each with their own
23/// pros and cons.
24///
25/// For example, WASAPI is the standard audio host API that ships with the Windows operating
26/// system. However, due to historical limitations with respect to performance and flexibility,
27/// Steinberg created the ASIO API providing better audio device support for pro audio and
28/// low-latency applications. As a result, it is common for some devices and device capabilities to
29/// only be available via ASIO, while others are only available via WASAPI.
30///
31/// Another great example is the Linux platform. While the ALSA host API is the lowest-level API
32/// available to almost all distributions of Linux, its flexibility is limited as it requires that
33/// each process have exclusive access to the devices with which they establish streams. PulseAudio
34/// is another popular host API that aims to solve this issue by providing user-space mixing,
35/// however it has its own limitations w.r.t. low-latency and high-performance audio applications.
36/// JACK is yet another host API that is more suitable to pro-audio applications, however it is
37/// less readily available by default in many Linux distributions and is known to be tricky to
38/// set up.
39///
40/// [`Host`]: crate::Host
41pub trait HostTrait {
42    /// The type used for enumerating available devices by the host.
43    type Devices: Iterator<Item = Self::Device>;
44    /// The `Device` type yielded by the host.
45    type Device: DeviceTrait;
46
47    /// Whether or not the host is available on the system.
48    fn is_available() -> bool;
49
50    /// An iterator yielding all [`Device`](DeviceTrait)s currently available to the host on the system.
51    ///
52    /// Can be empty if the system does not support audio in general.
53    ///
54    /// # Errors
55    ///
56    /// - [`ErrorKind::HostUnavailable`] if the host has become unreachable (e.g. the audio
57    ///   daemon crashed or was stopped).
58    /// - [`ErrorKind::BackendError`] for unclassifiable backend failures.
59    ///
60    /// [`ErrorKind::HostUnavailable`]: crate::ErrorKind::HostUnavailable
61    /// [`ErrorKind::BackendError`]: crate::ErrorKind::BackendError
62    fn devices(&self) -> Result<Self::Devices, Error>;
63
64    /// Fetches a [`Device`](DeviceTrait) based on a [`DeviceId`] if available
65    ///
66    /// Returns `None` if no device matching the id is found
67    fn device_by_id(&self, id: &DeviceId) -> Option<Self::Device> {
68        self.devices()
69            .ok()?
70            .find(|device| device.id().ok().as_ref() == Some(id))
71    }
72
73    /// The default input audio device on the system.
74    ///
75    /// Returns `None` if no input device is available.
76    ///
77    /// Some backends reroute a stream built from this device to the new default device when it
78    /// changes; capture continues there, and [`ErrorKind::DeviceChanged`] is reported. Other
79    /// backends report [`ErrorKind::DeviceNotAvailable`] instead, and the caller must rebuild the
80    /// stream.
81    ///
82    /// [`ErrorKind::DeviceChanged`]: crate::ErrorKind::DeviceChanged
83    /// [`ErrorKind::DeviceNotAvailable`]: crate::ErrorKind::DeviceNotAvailable
84    fn default_input_device(&self) -> Option<Self::Device>;
85
86    /// The default output audio device on the system.
87    ///
88    /// Returns `None` if no output device is available.
89    ///
90    /// Some backends reroute a stream built from this device to the new default device when it
91    /// changes; playback continues there, and [`ErrorKind::DeviceChanged`] is reported. Other
92    /// backends report [`ErrorKind::DeviceNotAvailable`] instead, and the caller must rebuild the
93    /// stream.
94    ///
95    /// [`ErrorKind::DeviceChanged`]: crate::ErrorKind::DeviceChanged
96    /// [`ErrorKind::DeviceNotAvailable`]: crate::ErrorKind::DeviceNotAvailable
97    fn default_output_device(&self) -> Option<Self::Device>;
98
99    /// An iterator yielding all `Device`s currently available to the system that support one or more
100    /// input stream formats.
101    ///
102    /// Can be empty if the system does not support audio input.
103    ///
104    /// # Errors
105    ///
106    /// Propagates errors from [`devices`](Self::devices).
107    fn input_devices(&self) -> Result<InputDevices<Self::Devices>, Error> {
108        Ok(self.devices()?.filter(DeviceTrait::supports_input))
109    }
110
111    /// An iterator yielding all `Device`s currently available to the system that support one or more
112    /// output stream formats.
113    ///
114    /// Can be empty if the system does not support audio output.
115    ///
116    /// # Errors
117    ///
118    /// Propagates errors from [`devices`](Self::devices).
119    fn output_devices(&self) -> Result<OutputDevices<Self::Devices>, Error> {
120        Ok(self.devices()?.filter(DeviceTrait::supports_output))
121    }
122}

124/// A device that is capable of audio input and/or output.
125///
126/// Please note that `Device`s may become invalid if they get disconnected. Therefore, all the
127/// methods that involve a device return a `Result` allowing the user to handle this case.
128pub trait DeviceTrait: PartialEq + Eq + Hash + Debug + Display {
129    /// The iterator type yielding supported input stream formats.
130    type SupportedInputConfigs: Iterator<Item = SupportedStreamConfigRange>;
131    /// The iterator type yielding supported output stream formats.
132    type SupportedOutputConfigs: Iterator<Item = SupportedStreamConfigRange>;
133    /// The stream type created by [`build_input_stream_raw`] and [`build_output_stream_raw`].
134    ///
135    /// [`build_input_stream_raw`]: Self::build_input_stream_raw
136    /// [`build_output_stream_raw`]: Self::build_output_stream_raw
137    type Stream: StreamTrait;
138
139    /// Structured description of the device with metadata.
140    ///
141    /// This returns a [`DeviceDescription`] containing structured information about the device,
142    /// including name, manufacturer (if available), device type, bus type, and other
143    /// platform-specific metadata.
144    ///
145    /// For the device name as a string, use `device.to_string()` or format it with `{}`. For the
146    /// full structured description, call `device.description()?` and format or inspect that.
147    ///
148    /// # Errors
149    ///
150    /// - [`ErrorKind::DeviceNotAvailable`] if the device has been disconnected.
151    ///
152    /// [`ErrorKind::DeviceNotAvailable`]: crate::ErrorKind::DeviceNotAvailable
153    fn description(&self) -> Result<DeviceDescription, Error>;
154
155    /// The ID of the device.
156    ///
157    /// This ID uniquely identifies the device on the host. It should be stable across program
158    /// runs, device disconnections, and system reboots where possible.
159    ///
160    /// # Errors
161    ///
162    /// - [`ErrorKind::DeviceNotAvailable`] if the device has been disconnected.
163    ///
164    /// [`ErrorKind::DeviceNotAvailable`]: crate::ErrorKind::DeviceNotAvailable
165    fn id(&self) -> Result<DeviceId, Error>;
166
167    /// True if the device supports audio input, otherwise false
168    fn supports_input(&self) -> bool {
169        self.supported_input_configs()
170            .is_ok_and(|mut iter| iter.next().is_some())
171    }
172
173    /// True if the device supports audio output, otherwise false
174    fn supports_output(&self) -> bool {
175        self.supported_output_configs()
176            .is_ok_and(|mut iter| iter.next().is_some())
177    }
178
179    /// An iterator yielding input stream configurations that are supported by the device.
180    ///
181    /// # Errors
182    ///
183    /// - [`ErrorKind::DeviceNotAvailable`] if the device has been disconnected.
184    /// - [`ErrorKind::UnsupportedConfig`] if the backend could not determine what the device
185    ///   supports.
186    /// - [`ErrorKind::UnsupportedOperation`] if the device does not support input.
187    ///
188    /// [`ErrorKind::DeviceNotAvailable`]: crate::ErrorKind::DeviceNotAvailable
189    /// [`ErrorKind::UnsupportedConfig`]: crate::ErrorKind::UnsupportedConfig
190    /// [`ErrorKind::UnsupportedOperation`]: crate::ErrorKind::UnsupportedOperation
191    fn supported_input_configs(&self) -> Result<Self::SupportedInputConfigs, Error>;
192
193    /// An iterator yielding output stream configurations that are supported by the device.
194    ///
195    /// # Errors
196    ///
197    /// - [`ErrorKind::DeviceNotAvailable`] if the device has been disconnected.
198    /// - [`ErrorKind::UnsupportedConfig`] if the backend could not determine what the device
199    ///   supports.
200    /// - [`ErrorKind::UnsupportedOperation`] if the device does not support output.
201    ///
202    /// [`ErrorKind::DeviceNotAvailable`]: crate::ErrorKind::DeviceNotAvailable
203    /// [`ErrorKind::UnsupportedConfig`]: crate::ErrorKind::UnsupportedConfig
204    /// [`ErrorKind::UnsupportedOperation`]: crate::ErrorKind::UnsupportedOperation
205    fn supported_output_configs(&self) -> Result<Self::SupportedOutputConfigs, Error>;
206
207    /// The default input stream configuration for the device.
208    ///
209    /// # Errors
210    ///
211    /// - [`ErrorKind::DeviceNotAvailable`] if the device has been disconnected.
212    /// - [`ErrorKind::UnsupportedConfig`] if the device has no default input configuration.
213    /// - [`ErrorKind::UnsupportedOperation`] if the device does not support input.
214    ///
215    /// [`ErrorKind::DeviceNotAvailable`]: crate::ErrorKind::DeviceNotAvailable
216    /// [`ErrorKind::UnsupportedConfig`]: crate::ErrorKind::UnsupportedConfig
217    /// [`ErrorKind::UnsupportedOperation`]: crate::ErrorKind::UnsupportedOperation
218    fn default_input_config(&self) -> Result<SupportedStreamConfig, Error>;
219
220    /// The default output stream configuration for the device.
221    ///
222    /// # Errors
223    ///
224    /// - [`ErrorKind::DeviceNotAvailable`] if the device has been disconnected.
225    /// - [`ErrorKind::UnsupportedConfig`] if the device has no default output configuration.
226    /// - [`ErrorKind::UnsupportedOperation`] if the device does not support output.
227    ///
228    /// [`ErrorKind::DeviceNotAvailable`]: crate::ErrorKind::DeviceNotAvailable
229    /// [`ErrorKind::UnsupportedConfig`]: crate::ErrorKind::UnsupportedConfig
230    /// [`ErrorKind::UnsupportedOperation`]: crate::ErrorKind::UnsupportedOperation
231    fn default_output_config(&self) -> Result<SupportedStreamConfig, Error>;
232
233    /// Create an input stream.
234    ///
235    /// # Parameters
236    ///
237    /// * `config` - The stream configuration including sample rate, channels, and buffer size.
238    /// * `data_callback` - Called periodically with captured audio data. The callback receives
239    ///   a slice of samples in the format `T` and timing information.
240    /// * `error_callback` - Called when a stream error occurs (e.g., device disconnected).
241    /// * `timeout` - Time to wait for the backend to initialize the stream. `None` waits
242    ///   indefinitely; `Some(duration)` limits how long to wait. Note: not all backends honor
243    ///   this value.
244    ///
245    /// # Errors
246    ///
247    /// - [`ErrorKind::UnsupportedConfig`] if the sample rate, channel count, buffer size, or
248    ///   sample format is not supported by the device.
249    /// - [`ErrorKind::UnsupportedOperation`] if the device does not support input streams.
250    /// - [`ErrorKind::DeviceNotAvailable`] if the device has been disconnected.
251    /// - [`ErrorKind::DeviceBusy`] if the device is temporarily in use by another application.
252    /// - [`ErrorKind::PermissionDenied`] if the process lacks permission to access the device.
253    /// - [`ErrorKind::InvalidInput`] if the configuration parameters are invalid.
254    ///
255    /// [`ErrorKind::UnsupportedConfig`]: crate::ErrorKind::UnsupportedConfig
256    /// [`ErrorKind::UnsupportedOperation`]: crate::ErrorKind::UnsupportedOperation
257    /// [`ErrorKind::DeviceNotAvailable`]: crate::ErrorKind::DeviceNotAvailable
258    /// [`ErrorKind::DeviceBusy`]: crate::ErrorKind::DeviceBusy
259    /// [`ErrorKind::PermissionDenied`]: crate::ErrorKind::PermissionDenied
260    /// [`ErrorKind::InvalidInput`]: crate::ErrorKind::InvalidInput
261    fn build_input_stream<T, D, E>(
262        &self,
263        config: StreamConfig,
264        mut data_callback: D,
265        error_callback: E,
266        timeout: Option<Duration>,
267    ) -> Result<Self::Stream, Error>
268    where
269        T: SizedSample,
270        D: FnMut(&[T], &InputCallbackInfo) + Send + 'static,
271        E: FnMut(Error) + Send + 'static,
272    {
273        self.build_input_stream_raw(
274            config,
275            T::FORMAT,
276            move |data, info| {
277                data_callback(
278                    data.as_slice()
279                        .expect("host supplied incorrect sample type"),
280                    info,
281                )
282            },
283            error_callback,
284            timeout,
285        )
286    }

288    /// Create an output stream.
289    ///
290    /// # Parameters
291    ///
292    /// * `config` - The stream configuration including sample rate, channels, and buffer size.
293    /// * `data_callback` - Called periodically to fill the output buffer. The callback receives
294    ///   a mutable slice of samples in the format `T` to be filled with audio data, along with

297    /// * `error_callback` - Called when a stream error occurs (e.g., device disconnected).
298    /// * `timeout` - Time to wait for the backend to initialize the stream. `None` waits
299    ///   indefinitely; `Some(duration)` limits how long to wait. Note: not all backends honor
300    ///   this value.
301    ///
302    /// # Errors
303    ///
304    /// - [`ErrorKind::UnsupportedConfig`] if the sample rate, channel count, buffer size, or
305    ///   sample format is not supported by the device.
306    /// - [`ErrorKind::UnsupportedOperation`] if the device does not support output streams.
307    /// - [`ErrorKind::DeviceNotAvailable`] if the device has been disconnected.
308    /// - [`ErrorKind::DeviceBusy`] if the device is temporarily in use by another application.
309    /// - [`ErrorKind::PermissionDenied`] if the process lacks permission to access the device.
310    /// - [`ErrorKind::InvalidInput`] if the configuration parameters are invalid.
311    ///
312    /// [`ErrorKind::UnsupportedConfig`]: crate::ErrorKind::UnsupportedConfig
313    /// [`ErrorKind::UnsupportedOperation`]: crate::ErrorKind::UnsupportedOperation
314    /// [`ErrorKind::DeviceNotAvailable`]: crate::ErrorKind::DeviceNotAvailable

338                )
339            },
340            error_callback,
341            timeout,
342        )
343    }

436/// A stream created from [`Device`](DeviceTrait), with methods to control it.
437pub trait StreamTrait {
438    /// Start the stream.
439    ///
440    /// Streams returned by `build_*_stream` are always stopped, so `play` must be called before the
441    /// data callback will fire. Despite the name, this applies equally to input (capture) streams.
442    ///
443    /// # Errors
444    ///
445    /// - [`ErrorKind::DeviceNotAvailable`] if the device has been disconnected.
446    /// - [`ErrorKind::StreamInvalidated`] if the stream configuration has changed and the stream
447    ///   must be rebuilt.
448    ///
449    /// [`ErrorKind::DeviceNotAvailable`]: crate::ErrorKind::DeviceNotAvailable
450    /// [`ErrorKind::StreamInvalidated`]: crate::ErrorKind::StreamInvalidated
451    fn play(&self) -> Result<(), Error>;
452
453    /// Pause the stream. Some devices support suspending at the hardware level (saving energy);
454    /// others stop only the data callback while the hardware keeps running.
455    ///
456    /// Note: Not all devices support suspending the stream at the hardware level. This method may
457    /// fail in these cases.
458    ///
459    /// # Errors
460    ///
461    /// - [`ErrorKind::UnsupportedOperation`] if the backend does not support pausing streams.
462    /// - [`ErrorKind::DeviceNotAvailable`] if the device has been disconnected.
463    /// - [`ErrorKind::StreamInvalidated`] if the stream configuration has changed and the stream
464    ///   must be rebuilt.
465    ///
466    /// [`ErrorKind::UnsupportedOperation`]: crate::ErrorKind::UnsupportedOperation
467    /// [`ErrorKind::DeviceNotAvailable`]: crate::ErrorKind::DeviceNotAvailable
468    /// [`ErrorKind::StreamInvalidated`]: crate::ErrorKind::StreamInvalidated
469    fn pause(&self) -> Result<(), Error>;
470
471    /// Returns the backend's best available estimate of the number of frames per callback.
472    ///
473    /// The value is available immediately after stream creation: for fixed buffer sizes this is
474    /// the negotiated hardware size; for default buffer sizes this is the backend's configured
475    /// default. The value is updated when it changes during the lifetime of the stream.
476    ///
477    /// # Errors
478    ///
479    /// - [`ErrorKind::UnsupportedOperation`] if the backend cannot query the buffer size.
480    /// - [`ErrorKind::BackendError`] for unclassifiable backend failures.
481    ///
482    /// # Implementation notes
483    ///
484    /// It is not enforced that each callback delivers exactly this many frames. The actual frame
485    /// count for each callback is given by its buffer.
486    ///
487    /// `buffer_size()` is primarily intended for sizing pre-allocated buffers, but must not be
488    /// trusted as a guaranteed bound. An incorrect implementation of `buffer_size()` should not
489    /// lead to memory safety violations.
490    ///
491    /// [`ErrorKind::UnsupportedOperation`]: crate::ErrorKind::UnsupportedOperation
492    /// [`ErrorKind::BackendError`]: crate::ErrorKind::BackendError
493    fn buffer_size(&self) -> Result<crate::FrameCount, Error>;
494
495    /// Returns a [`StreamInstant`] representing the current moment on the stream's clock.
496    ///
497    /// The clock is **monotonic**: successive calls to `now()` will never return a value earlier
498    /// than a previous one, and the returned value will never be earlier than any `callback`,
499    /// `capture`, or `playback` instant already delivered to the stream's data callback.
500    ///
501    /// The returned value shares the same time base as the [`StreamInstant`]s delivered to the
502    /// stream's data callback via [`crate::InputStreamTimestamp::callback`] and
503    /// [`crate::OutputStreamTimestamp::callback`], so durations between them are meaningful.
504    fn now(&self) -> StreamInstant;
505}

507/// Compile-time assertion that a stream type implements [`Send`].
508///
509/// Custom host implementations should use this macro to verify their `Stream` type
510/// can be safely transferred between threads, as required by CPAL's API.
511///
512/// # Example
513///
514/// ```
515/// use cpal::assert_stream_send;
516/// struct MyStream { /* ... */ }
517/// assert_stream_send!(MyStream);
518/// ```
519#[macro_export]
520macro_rules! assert_stream_send {
521    ($t:ty) => {
522        const fn _assert_stream_send<T: Send>() {}
523        const _: () = _assert_stream_send::<$t>();
524    };
525}
