4/// Engine-clock ticks.
5/// Zero is invalid. Convert using the clock's `QueryPerformanceFrequency()`
6/// value in ticks per second; no fixed frequency is assumed. Conversions and
7/// arithmetic accept zero ticks but do not make that value valid.
8///
9/// SDK: [EngineTime](https://github.com/WopsS/RED4ext.SDK/blob/2216befc23924b46996a9335043aebb31ba6a757/include/RED4ext/Scripting/Natives/EngineTime.hpp).
10#[derive(Debug, Default, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash)]
11#[repr(transparent)]
12pub struct EngineTime(u64);
13
14impl EngineTime {
15    /// Returns whether the tick count is nonzero.
16    pub fn is_valid(&self) -> bool {
17        self.0 != 0
18    }
19
20    /// Wraps a raw tick count, including the invalid zero value.
21    pub const fn from_ticks(ticks: u64) -> Self {
22        Self(ticks)
23    }
24
25    /// Returns the raw tick count.
26    pub const fn ticks(self) -> u64 {
27        self.0
28    }
29
30    /// Converts to seconds, potentially losing precision for large tick counts.
31    pub fn as_secs_f64(self, frequency: NonZeroU64) -> f64 {
32        self.0 as f64 / frequency.get() as f64
33    }
34
35    /// Converts nonnegative, finite seconds, truncating fractional ticks.
36    ///
37    /// # Errors
38    /// Returns `EngineTimeError::NotANumber` for NaN, or
39    /// `EngineTimeError::OutOfBounds` for negative, infinite, or overflowing values.
40    pub fn from_secs_f64(seconds: f64, frequency: NonZeroU64) -> Result<Self, EngineTimeError> {
41        if seconds.is_nan() {
42            return Err(EngineTimeError::NotANumber);
43        }
44        let ticks = seconds * frequency.get() as f64;
45        // u64::MAX rounds up to 2^64 in f64, which must be rejected.
46        if seconds < 0.0 || !ticks.is_finite() || ticks >= 18_446_744_073_709_551_616.0 {
47            return Err(EngineTimeError::OutOfBounds);
48        }
49        Ok(Self(ticks as u64))
50    }
51
52    /// Converts a duration using integer arithmetic, truncating fractional ticks.
53    ///
54    /// # Errors
55    /// Returns `EngineTimeError::OutOfBounds` if the tick count exceeds `u64::MAX`.
56    pub fn from_duration(
57        duration: Duration,
58        frequency: NonZeroU64,
59    ) -> Result<Self, EngineTimeError> {
60        let frequency = u128::from(frequency.get());
61        let ticks = u128::from(duration.as_secs()) * frequency
62            + u128::from(duration.subsec_nanos()) * frequency / 1_000_000_000;
63        u64::try_from(ticks)
64            .map(Self)
65            .map_err(|_| EngineTimeError::OutOfBounds)
66    }
67
68    /// Converts to a duration, truncating fractional nanoseconds.
69    pub fn to_duration(self, frequency: NonZeroU64) -> Duration {
70        let seconds = self.0 / frequency.get();
71        let nanos =
72            u128::from(self.0 % frequency.get()) * 1_000_000_000 / u128::from(frequency.get());
73        Duration::new(seconds, nanos as u32)
74    }
75
76    /// Adds a duration after truncating it to whole ticks.
77    /// Returns `None` if duration conversion or addition overflows.
78    pub fn checked_add_duration(self, duration: Duration, frequency: NonZeroU64) -> Option<Self> {
79        self.0
80            .checked_add(Self::from_duration(duration, frequency).ok()?.0)
81            .map(Self)
82    }
83
84    /// Subtracts a duration after truncating it to whole ticks.
85    /// Returns `None` if duration conversion overflows or subtraction underflows.
86    pub fn checked_sub_duration(self, duration: Duration, frequency: NonZeroU64) -> Option<Self> {
87        self.0
88            .checked_sub(Self::from_duration(duration, frequency).ok()?.0)
89            .map(Self)
90    }
91}
92
93/// Failure to convert seconds or a duration into engine-clock ticks.
94#[derive(Debug)]
95pub enum EngineTimeError {
96    /// The input is negative, infinite, or too large for an unsigned 64-bit tick count.
97    OutOfBounds,
98    /// The floating-point seconds value is NaN.
99    NotANumber,
100}
101
102impl std::fmt::Display for EngineTimeError {
103    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
104        write!(
105            f,
106            "{}",
107            match self {
108                Self::OutOfBounds => "engine time must fit in unsigned 64-bit ticks",
109                Self::NotANumber => "invalid NaN",
110            }
111        )
112    }
113}
114
115impl std::error::Error for EngineTimeError {}

117#[cfg(test)]
118mod tests {
119    use std::time::Duration;
120
121    use super::EngineTime;
122
123    #[test]
124    fn layout_and_ticks() {
125        assert_eq!(std::mem::size_of::<EngineTime>(), 8);
126        assert_eq!(
127            std::mem::align_of::<EngineTime>(),
128            std::mem::align_of::<u64>()
129        );
130        assert!(!EngineTime::default().is_valid());
131        assert_eq!(EngineTime::from_ticks(u64::MAX).ticks(), u64::MAX);
132    }
133
134    #[test]
135    fn frequency_aware_conversions() {
136        let frequency = std::num::NonZeroU64::new(10_000_000).unwrap();
137        let time = EngineTime::from_duration(Duration::from_millis(3200), frequency).unwrap();
138        assert_eq!(time.ticks(), 32_000_000);
139        assert_eq!(time.as_secs_f64(frequency), 3.2);
140        assert_eq!(time.to_duration(frequency), Duration::from_millis(3200));
141        let after = time
142            .checked_add_duration(Duration::from_millis(4100), frequency)
143            .unwrap();
144        assert_eq!(after.ticks(), 73_000_000);
145        assert_eq!(
146            after.checked_sub_duration(Duration::from_millis(4100), frequency),
147            Some(time)
148        );
149        let thirds = std::num::NonZeroU64::new(3).unwrap();
150        assert_eq!(EngineTime::from_secs_f64(0.5, thirds).unwrap().ticks(), 1);
151        assert_eq!(
152            EngineTime::from_ticks(1).to_duration(thirds),
153            Duration::from_nanos(333_333_333)
154        );
155    }
156
157    #[test]
158    fn bounds() {
159        let frequency = std::num::NonZeroU64::new(1).unwrap();
160        for seconds in [
161            f64::NAN,
162            f64::INFINITY,
163            f64::NEG_INFINITY,
164            -1.0,
165            18_446_744_073_709_551_616.0,
166        ] {
167            assert!(EngineTime::from_secs_f64(seconds, frequency).is_err());
168        }
169        assert_eq!(
170            EngineTime::from_duration(Duration::from_secs(u64::MAX), frequency)
171                .unwrap()
172                .ticks(),
173            u64::MAX
174        );
175        assert!(
176            EngineTime::from_duration(Duration::MAX, std::num::NonZeroU64::new(u64::MAX).unwrap())
177                .is_err()
178        );
179        assert!(
180            EngineTime::from_ticks(u64::MAX)
181                .checked_add_duration(Duration::from_secs(1), frequency)
182                .is_none()
183        );
184        assert!(
185            EngineTime::default()
186                .checked_sub_duration(Duration::from_secs(1), frequency)
187                .is_none()
188        );
