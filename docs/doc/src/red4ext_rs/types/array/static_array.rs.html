5/// Inline fixed-capacity array with a logical element count.
6///
7/// Conversion from `[T; N]` fills the array. [`NativeRepr`] names are supported
8/// only for capacities 1 through 8.
9///
10/// SDK: [array RTTI](https://github.com/WopsS/RED4ext.SDK/blob/2216befc23924b46996a9335043aebb31ba6a757/include/RED4ext/RTTITypes.hpp).
11#[derive(Debug)]
12#[repr(C)]
13pub struct StaticArray<T, const N: usize> {
14    entries: [T; N],
15    size: u32,
16}
17
18const fn const_digit_str<const N: usize>() -> &'static str {
19    match N {
20        1 => "1",
21        2 => "2",
22        3 => "3",
23        4 => "4",
24        5 => "5",
25        6 => "6",
26        7 => "7",
27        8 => "8",
28        _ => unimplemented!(),
29    }
30}
31
32unsafe impl<T: NativeRepr, const N: usize> NativeRepr for StaticArray<T, N> {
33    const NAME: &'static str = combine!(
34        combine!(combine!("[", const_digit_str::<N>()), "]"),
35        T::NAME
36    );
37}
38
39impl<T, const N: usize> From<[T; N]> for StaticArray<T, N> {
40    fn from(entries: [T; N]) -> Self {
41        Self {
42            size: entries.len() as u32,
43            entries,
44        }
45    }
46}
47
48impl<T, const N: usize> StaticArray<T, N> {
49    /// Returns the elements within the logical size.
50    #[inline]
51    pub fn entries(&self) -> &[T] {
52        &self.entries[..self.size as usize]
53    }
54
55    /// Returns the logical number of elements, not the capacity `N`.
56    #[inline]
57    pub fn size(&self) -> u32 {
58        self.size
59    }
60}
