33/// Native game instance,
34/// not REDscript's [`GameInstance`]. Access through [`GameEngine::game_instance`].
35/// This is a borrowed engine object, not a constructible or owning Rust value.
36///
37/// SDK: [GameInstance](https://github.com/WopsS/RED4ext.SDK/blob/2216befc23924b46996a9335043aebb31ba6a757/include/RED4ext/GameEngine.hpp).
38#[repr(transparent)]
39pub struct NativeGameInstance(red::GameInstance);
40
41impl NativeGameInstance {
42    /// Looks up a system by its RTTI type and acquires a strong reference.
43    ///
44    /// Returns a null [`Ref`] if the system is absent or its weak reference has expired.
45    #[inline]
46    pub fn get_system(&self, ty: &Type) -> Ref<ScriptableSystem> {
47        let instance = unsafe { (self.vft().get_system)(self, ty) };
48        if instance.is_null() {
49            return Ref::default();
50        }
51        let instance: &WeakRef<ScriptableSystem> =
52            unsafe { mem::transmute(&(*instance)._base.ref_) };
53        instance.clone().upgrade().unwrap_or_default()
54    }
55
56    #[inline]
57    fn vft(&self) -> &GameInstanceVft {
58        unsafe { &*(self.0.vtable_ as *const GameInstanceVft) }
59    }
60}

86/// SDK: [CGameEngine](https://github.com/WopsS/RED4ext.SDK/blob/2216befc23924b46996a9335043aebb31ba6a757/include/RED4ext/GameEngine.hpp).
87pub struct GameEngine(red::CGameEngine);
88
89impl GameEngine {
90    /// Borrows the engine singleton, which must already be initialized and remain live.
91    pub fn get<'a>() -> &'a Self {
92        unsafe { &*(red::CGameEngine::Get() as *const GameEngine) }
93    }
94
95    /// Borrows the native game instance from the initialized game framework.
96    ///
97    /// The framework and game instance must be live; this method does not check for null.
98    pub fn game_instance(&self) -> &NativeGameInstance {
99        unsafe { &*((*self.0.framework).gameInstance as *const NativeGameInstance) }
100    }
