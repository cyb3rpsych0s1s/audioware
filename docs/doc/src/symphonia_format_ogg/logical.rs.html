22#[derive(Copy, Clone, Debug)]
23struct Bound {
24    /// The page sequence number.
25    seq: u32,
26    /// Indicates if this is the last page.
27    is_last_page: bool,
28    /// The start or end timestamp (depends on the type of bound).
29    ts: Timestamp,
30    /// The samples to discard from the start or end (depends on type of bound).
31    discard: Duration,
32}

42}

108    /// Read a page. Specifying whether this is the initial bitstream page.
109    pub fn read_page_init(&mut self, page: &Page<'_>, is_init_page: bool) -> Result<Vec<SideData>> {
110        // Side data vector. This will not allocate unless data is pushed to it (normal case).
111        let mut side_data = Vec::new();
112
113        // If the last sequence number is available, detect non-monotonicity and discontinuities
114        // in the stream. In these cases, clear any partial packet data.
115        if let Some(last_ts) = &self.prev_page_info {
116            if page.header.sequence < last_ts.seq {
117                warn!("detected stream page non-monotonicity");
118                self.part_len = 0;
119            }
120            else if page.header.sequence - last_ts.seq > 1 {
121                warn!(
122                    "detected stream discontinuity of {} page(s)",
123                    page.header.sequence - last_ts.seq
124                );
125                self.part_len = 0;
126            }
127        }
128
129        // Keep a copy of the previous page information.
130        let prev_page_info = self.prev_page_info.take();
131
132        // Update with new page information.
133        self.prev_page_info = Some(PageInfo {
134            seq: page.header.sequence,
135            is_last_page: page.header.is_last_page,
136            // Propagate the last known valid bitstream timestamp. This will be updated later if
137            // a bitstream packet ends in this page.
138            end_ts: prev_page_info.and_then(|info| info.end_ts),
139        });
140
141        // If there is partial packet data buffered, a continuation page is expected.
142        if !page.header.is_continuation && self.part_len > 0 {
143            warn!("expected a continuation page");
144
145            // Clear partial packet data.
146            self.part_len = 0;
147        }
148
149        let mut iter = page.packets();
150
151        // If there is no partial packet data buffered, a continuation page is not expected.
152        if page.header.is_continuation && self.part_len == 0 {
153            // If the continuation page contains packets, drop the first packet since it would
154            // require partial packet data to be complete. Otherwise, ignore this page entirely.
155            if page.num_packets() > 0 {
156                warn!("unexpected continuation page, ignoring incomplete first packet");
157                iter.next();
158            }
159            else {
160                warn!("unexpected continuation page, ignoring page");
161                return Ok(side_data);
162            }
163        }
164
165        let mut total_pkt_dur = Duration::ZERO;
166        let mut total_pkt_discard = Duration::ZERO;
167
168        let num_prev_packets = self.packets.len();
169
170        for buf in &mut iter {
171            // Get a packet with data from the partial packet buffer, the page, or both.
172            let data = self.get_packet(buf);
173
174            // Perform packet mapping. If the packet contains stream data, queue it onto the packet
175            // queue. If it contains side data, then add it to the side data list. Ignore other
176            // types of packet data.
177            match self.mapper.map_packet(&data) {
178                Ok(MapResult::StreamData { dur, discard }) => {
179                    total_pkt_dur = total_pkt_dur.saturating_add(dur);
180                    total_pkt_discard = total_pkt_discard.saturating_add(discard);
181
182                    let packet = PacketBuilder::new()
183                        .track_id(page.header.serial)
184                        .pts(Timestamp::ZERO)
185                        .dur(dur)
186                        .data(data)
187                        .trim_start(discard)
188                        .build();
189
190                    self.packets.push_back(packet);
191                }
192                Ok(MapResult::SideData { data }) => side_data.push(data),
193                Err(e) => {
194                    warn!("mapping packet failed ({e}), skipping")
195                }
196                _ => (),
197            }
198        }
199
200        // If the page contains partial packet data, then save the partial packet data for later
201        // as the packet will be completed on a later page.
202        if let Some(buf) = iter.partial_packet() {
203            self.save_partial_packet(buf)?;
204        }
205
206        // If one or more bitstream packets ended in this page, process them.
207        if num_prev_packets < self.packets.len() {
208            // The page's timestamp is one past the last valid frame in the last completed packet
209            // in this page.
210            let page_end_ts = self.mapper.absgp_to_ts(page.header.absgp);
211
212            // Update previous page information with the timestamp.
213            let prev_page = self
214                .prev_page_info
215                .as_mut()
216                .expect("prev_page_info unconditionally assigned above");
217            prev_page.end_ts = Some(page_end_ts);
218
219            // Compute the page's start timestamp.
220            let page_start_ts = if let Some(ts) = prev_page_info.and_then(|prev| prev.end_ts) {
221                // The previous page is known and it has a valid end timestamp. Use it as this
222                // page's start timestamp.
223                ts
224            }
225            else {
226                let is_single_page_stream =
227                    page.header.is_last_page && (is_init_page || self.is_single_page_stream());
228
229                // The lower-bound for the page's start timestamp.
230                let page_start_ts_raw = page_end_ts.saturating_sub(total_pkt_dur);
231
232                if is_single_page_stream {
233                    // In a single-page stream that begins at t = 0, the page end timestamp is the
234                    // exact length of valid (no delay or padding) frames. Therefore, the page start
235                    // timestamp is known to be equal to -delay frames. All additional frames are
236                    // padding.
237                    //
238                    // If the stream does not start at t = 0, then it is not possible to determine
239                    // whether frames should be discarded from the start or the end of the stream.
240                    // Therefore, we assume t = 0 for the remainder.
241                    let total_dur_no_padding =
242                        match total_pkt_discard.checked_add((page_end_ts.get() as u64).into()) {
243                            Some(v) => v,
244                            None => return decode_error("ogg: duration overflow"),
245                        };
246
247                    if total_pkt_dur >= total_dur_no_padding {
248                        // The total packet duration is >= delay + valid (assuming t = 0) frames,
249                        // the extra frames must be padding.
250                        //
251                        // If the assumption the stream begins at t = 0 is false, then the following
252                        // misbehaviour will occur:
253                        //
254                        // If the encoder set t < 0 to discard additional frames at the start of the
255                        // stream, then the additional frames will be discard from the end, not the
256                        // beginning.
257                        //
258                        // If the encoder set t > 0 to indicate the media begins later, then no
259                        // padding frames will get discarded.
260                        Timestamp::from(-(total_pkt_discard.get() as i64))
261                    }
262                    else {
263                        // Stream starts at t > 0.
264                        page_start_ts_raw
265                    }
266                }
267                else {
268                    // In a multi-page stream, all pages other than the last have no padding.
269                    // Therefore, the naive calculation is always valid because the total packet
270                    // duration would only include valid or discarded frames.
271                    //
272                    // For the last page, this calculation fails because the total packet duration
273                    // includes the padding and it is not possible to know how many frames are
274                    // valid. If the end bound is known, the correct page start timestamp can be
275                    // found.
276                    page_start_ts_raw
277                }
278            };
279
280            let mut next_pkt_pts = page_start_ts;
281
282            for packet in self.packets.iter_mut().skip(num_prev_packets) {
283                packet.pts = next_pkt_pts;
284
285                // The packet's duration is populated with the packet's decoded duration (includes
286                // delay and padding). Calculate the next packet's PTS. This is also the end PTS of
287                // the current packet.
288                next_pkt_pts = next_pkt_pts.saturating_add(packet.dur);
289
290                // Remove the start trim from the packet's duration. The start trim was populated
291                // earlier.
292                packet.dur = packet.dur.saturating_sub(packet.trim_start);
293
294                // If the end of the current packet exceeds the page end, trim the end. Don't trim
295                // more frames than available.
296                if next_pkt_pts > page_end_ts {
297                    packet.trim_end = page_end_ts.abs_delta(next_pkt_pts).min(packet.dur);
298                }
299
300                // Remove the end trim from the packet's duration.
301                packet.dur = packet.dur.saturating_sub(packet.trim_end);
302            }
303        }
304
305        Ok(side_data)
306    }

328    /// Examine the first page of the non-setup codec bitstream to obtain the start time and start
329    /// delay parameters.
330    pub fn inspect_start_page(&mut self, page: &Page<'_>) {
331        if self.start_bound.is_some() {
332            debug!("start page already found");
333            return;
334        }
335
336        let mut parser = match self.mapper.make_parser() {
337            Some(parser) => parser,
338            _ => {
339                debug!("failed to make start bound packet parser");
340                return;
341            }
342        };
343
344        // Sum the total duration of all packets, and the amount to discard.
345        let mut total_pkt_dur = Duration::ZERO;
346        let mut total_pkt_discard = Duration::ZERO;
347
348        for buf in page.packets() {
349            let (pkt_dur, pkt_discard) = parser.parse_next_packet_dur(buf);
350
351            // On overflow, it will not be possible to determine the start bound.
352            total_pkt_dur = match total_pkt_dur.checked_add(pkt_dur) {
353                Some(total) => total,
354                _ => return,
355            };
356            total_pkt_discard = match total_pkt_discard.checked_add(pkt_discard) {
357                Some(total) => total,
358                _ => return,
359            };
360        }
361
362        // Map the absolute granule position to a timestamp.
363        let page_end_ts = self.mapper.absgp_to_ts(page.header.absgp);
364
365        // For single page streams, the first page is also the last page. If the page is
366        // explicitly marked as the last page, then we assume the stream starts and ends on the
367        // same page. Therefore, we can only assume it starts at pts=0, and any discarded frames
368        // are padding frames, not delay frames.
369        let page_start_ts = if !page.header.is_last_page {
370            match page_end_ts.checked_sub(total_pkt_dur) {
371                Some(ts) => ts,
372                _ => return,
373            }
374        }
375        else {
376            Timestamp::new(-(total_pkt_discard.get() as i64))
377        };
378
379        let bound = Bound {
380            seq: page.header.sequence,
381            is_last_page: page.header.is_last_page,
382            ts: page_start_ts,
383            discard: total_pkt_discard,
384        };
385
386        // Update codec parameters.
387        let track = self.mapper.track_mut();
388
389        track.with_start_ts(bound.ts);
390
391        if bound.discard > Duration::ZERO {
392            track.with_delay(bound.discard.get() as u32);
393        }
394
395        // Update start bound.
396        self.start_bound = Some(bound);
397    }

399    /// Examines one or more of the last pages of the codec bitstream to obtain the end time and
400    /// end delay parameters. To obtain the end delay, at a minimum, the last two pages are
401    /// required. The state returned by each iteration of this function should be passed into the
402    /// subsequent iteration.
403    pub fn inspect_end_page(&mut self, state: &mut InspectState, page: &Page<'_>) {
404        // Do nothing if the end bound was found.
405        if self.end_bound.is_some() {
406            debug!("end page already found");
407            return;
408        }
409
410        // Get and/or create the packet parser.
411        let parser = match &mut state.parser {
412            Some(parser) => parser,
413            None => {
414                state.parser = self.mapper.make_parser();
415
416                let Some(parser) = &mut state.parser
417                else {
418                    debug!("failed to make end bound packet parser");
419                    return;
420                };
421
422                parser
423            }
424        };
425
426        // Calculate the page duration. Note that even though only the last page uses this duration,
427        // it is important to feed the packet parser so that the first packet of the final page
428        // doesn't have a duration of 0 due to lapping on some codecs.
429        let mut total_pkt_dur = Duration::ZERO;
430
431        for buf in page.packets() {
432            let (pkt_dur, _) = parser.parse_next_packet_dur(buf);
433
434            // On overflow it will be impossible to determine the end bound.
435            total_pkt_dur = match total_pkt_dur.checked_add(pkt_dur) {
436                Some(total) => total,
437                _ => return,
438            };
439        }
440
441        // Map the absolute granule position to a timestamp.
442        let page_end_ts = self.mapper.absgp_to_ts(page.header.absgp);
443
444        // The end delay can only be determined if this is the last page, and the timstamp of the
445        // second last page is known, or this is the only page of the stream.
446        let end_delay = if page.header.is_last_page {
447            if let Some(last_bound) = &state.bound {
448                // The actual ending timestamp of the decoded data is the timestamp of the previous
449                // page plus the decoded duration of this page. Subtract the stated page end
450                // timestamp to determine the padding. On overflow, it'll be impossible to determine
451                // the end delay, so force it to be 0.
452                last_bound
453                    .ts
454                    .checked_add(total_pkt_dur)
455                    .map(|actual_page_end_ts| actual_page_end_ts.abs_delta(page_end_ts))
456            }
457            else if self.start_bound.is_some_and(|b| b.seq == page.header.sequence) {
458                // The start and end page is the same page. The end delay is the amount of excess
459                // page duration after subtracting the page's end timestamp.
460
461                // The amount of start delay, if available.
462                let delay = self.start_bound.map(|s| s.discard).unwrap_or(Duration::ZERO);
463
464                // The page end timestamp does not include delay or padding. Therefore, it is also
465                // the duration of valid content assuming it is non-negative.
466                let valid = page_end_ts.duration_from(Timestamp::ZERO);
467
468                // The amount of valid content plus padding.
469                let valid_and_padding = total_pkt_dur.checked_sub(delay);
470
471                // Take the valid duration from the valid and padding duration to yield the end
472                // padding.
473                valid_and_padding
474                    .zip(valid)
475                    .and_then(|(valid_and_padding, valid)| valid_and_padding.checked_sub(valid))
476            }
477            else {
478                // Don't have the timestamp of the previous page so it is not possible to
479                // calculate the end delay.
480                None
481            }
482        }
483        else {
484            // Only the last page can have an end delay.
485            None
486        };
487
488        let bound = Bound {
489            seq: page.header.sequence,
490            is_last_page: page.header.is_last_page,
491            ts: page_end_ts,
492            discard: end_delay.unwrap_or(Duration::ZERO),
493        };
494
495        // If this is the last page, update the codec parameters.
496        if page.header.is_last_page {
497            // TODO: What if this is negative?
498            // TODO: Bounds are in timestamp units, not number of frames. Fix this up because it
499            // only works for audio codecs.
500            let num_frames = bound.ts.get() as u64;
501            let num_padding_frames = bound.discard.get() as u32;
502
503            let track = self.mapper.track_mut();
504
505            track.with_num_frames(num_frames);
506            track.with_duration(Duration::from(num_frames));
507
508            if num_padding_frames > 0 {
509                track.with_padding(num_padding_frames);
510            }
511
512            self.end_bound = Some(bound);
513        }
514
515        // Update the state's bound.
516        state.bound = Some(bound);
517    }

519    /// Examine a page in isolation and return the start and end timestamps as a tuple.
520    pub fn inspect_page(&mut self, page: &Page<'_>) -> (Timestamp, Timestamp) {
521        // Get the cumulative duration of all packets within this page.
522        let mut total_pkt_dur = Duration::ZERO;
523        let mut total_pkt_discard = Duration::ZERO;
524
525        if let Some(mut parser) = self.mapper.make_parser() {
526            for buf in page.packets() {
527                let (pkt_dur, pkt_discard) = parser.parse_next_packet_dur(buf);
528
529                total_pkt_dur = total_pkt_dur.saturating_add(pkt_dur);
530                total_pkt_discard = total_pkt_discard.saturating_add(pkt_discard);
531            }
532        }
533
534        let page_end_ts = self.mapper.absgp_to_ts(page.header.absgp);
535
536        // Map the absolute granule position to a timestamp.
537        let page_start_ts_raw = page_end_ts.saturating_sub(total_pkt_dur);
538
539        let page_start_ts = match self.start_bound {
540            // Start bound is known, and this is the first bitstream page.
541            Some(b) if b.seq == page.header.sequence => b.ts,
542            // Start bound is known, but this is not the first bitstream page.
543            Some(_) => page_start_ts_raw,
544            // Start bound is not known. Assume this is the first bitstream page.
545            None if page_start_ts_raw.is_negative() => {
546                Timestamp::new(-(total_pkt_discard.get() as i64))
547            }
548            // Start bound is not known, but the page start timestamp is positive. The
549            // stream likely started with a positive timestamp.
550            None => page_start_ts_raw,
551        };
552
553        (page_start_ts, page_end_ts)
554    }

556    /// Returns true if the stream is only a single page.
557    fn is_single_page_stream(&self) -> bool {
558        match self.start_bound {
559            // The starting page is known, and it is marked as the last page. This is a single-page
560            // stream.
561            Some(start) if start.is_last_page => true,
562            // The starting page is known, and it is not marked as the last page. This can be
563            // assumed to be a multi-page stream. NOTE: This assumption relies on the starting page
564            // being the page with first completed packet.
565            Some(start) => {
566                // However, if the starting and ending pages are known and have mismatched sequence
567                // numbers (this shouldn't happen), then the previous assumption was false.
568                match self.end_bound {
569                    Some(end) => start.seq == end.seq,
570                    _ => true,
571                }
572            }
573            // The starting page is not known, cannot assume this is a single-page stream.
574            _ => false,
575        }
576    }

578    fn get_packet(&mut self, packet_buf: &[u8]) -> Box<[u8]> {
579        if self.part_len == 0 {
580            Box::from(packet_buf)
581        }
582        else {
583            let mut buf = vec![0u8; self.part_len + packet_buf.len()];
584
585            // Split packet buffer into two portions: saved and new.
586            let (vec0, vec1) = buf.split_at_mut(self.part_len);
587
588            // Copy and consume the saved partial packet.
589            vec0.copy_from_slice(&self.part_buf[..self.part_len]);
590            self.part_len = 0;
591
592            // Read the remainder of the partial packet from the page.
593            vec1.copy_from_slice(packet_buf);
594
595            buf.into_boxed_slice()
596        }
597    }
598
599    fn save_partial_packet(&mut self, buf: &[u8]) -> Result<()> {
600        let new_part_len = self.part_len + buf.len();
601
602        if new_part_len > self.part_buf.len() {
603            // Do not exceed an a certain limit to prevent unbounded memory growth.
604            if new_part_len > LogicalStream::MAX_PACKET_LEN {
605                return decode_error("ogg: packet buffer would exceed max size");
606            }
607
608            // New partial packet buffer size, rounded up to the nearest 8K block.
609            let new_buf_len = (new_part_len + (8 * 1024 - 1)) & !(8 * 1024 - 1);
610            debug!("grow packet buffer to {new_buf_len} bytes");
611
612            self.part_buf.resize(new_buf_len, Default::default());
613        }
614
615        self.part_buf[self.part_len..new_part_len].copy_from_slice(buf);
616        self.part_len = new_part_len;
617
618        Ok(())
619    }
