8use symphonia_core::audio::{Channels, layouts};
9use symphonia_core::errors::{Result, decode_error, unsupported_error};
10use symphonia_core::io::{BufReader, FiniteStream, ReadBytes};
11
12/// Supported ALAC version.
13const ALAC_VERSION: u8 = 0;
14
15/// The ALAC "magic cookie" or codec specific configuration.
16#[derive(Debug, Clone)]
17pub struct MagicCookie {
18    pub frame_length: u32,
19    pub compatible_version: u8,
20    pub bit_depth: u8,
21    pub pb: u8,
22    pub mb: u8,
23    pub kb: u8,
24    pub num_channels: u8,
25    pub max_run: u16,
26    pub max_frame_bytes: u32,
27    pub avg_bit_rate: u32,
28    pub sample_rate: u32,
29    pub channels: Channels,
30}
31
32impl MagicCookie {
33    /// Read the ALAC magic cookie from the provided buffer.
34    pub fn read(mut buf: &[u8]) -> Result<MagicCookie> {
35        // The magic cookie must be atleast 24 bytes long.
36        if buf.len() < 24 {
37            return unsupported_error("common (alac): magic cookie size too small");
38        }
39
40        // The magic cookie may be preceeded by a FRMA atom. Skip over the FRMA atom.
41        if buf[4..8] == *b"frma" {
42            buf = &buf[12..];
43        }
44
45        // The magic cookie may be preceeded by an ALAC atom. Skip over the ALAC atom.
46        if buf[4..8] == *b"alac" {
47            buf = &buf[12..];
48        }
49
50        // The magic cookie must be either 24 or 48 bytes long.
51        if buf.len() != 24 && buf.len() != 48 {
52            return unsupported_error("common (alac): invalid magic cookie size");
53        }
54
55        let mut reader = BufReader::new(buf);
56
57        let mut config = MagicCookie {
58            frame_length: reader.read_be_u32()?,
59            compatible_version: reader.read_u8()?,
60            bit_depth: reader.read_u8()?,
61            pb: reader.read_u8()?,
62            mb: reader.read_u8()?,
63            kb: reader.read_u8()?,
64            num_channels: reader.read_u8()?,
65            max_run: reader.read_be_u16()?,
66            max_frame_bytes: reader.read_be_u32()?,
67            avg_bit_rate: reader.read_be_u32()?,
68            sample_rate: reader.read_be_u32()?,
69            channels: Default::default(),
70        };
71
72        // Only support up-to the currently implemented ALAC version.
73        if config.compatible_version > ALAC_VERSION {
74            return unsupported_error("common (alac): not compatible with alac version 0");
75        }
76
77        // A bit-depth greater than 32 is not allowed.
78        if config.bit_depth > 32 {
79            return decode_error("common (alac): invalid bit depth");
80        }
81
82        // Only 8 channel layouts exist.
83        // TODO: Support discrete/auxiliary channels.
84        if config.num_channels < 1 || config.num_channels > 8 {
85            return decode_error("common (alac): invalid number of channels");
86        }
87
88        // If the magic cookie is 48 bytes, the channel layout is explictly set, otherwise select a
89        // channel layout from the number of channels.
90        config.channels = if reader.byte_len() == 48 {
91            // The first field is the size of the channel layout info. This should always be 24.
92            if reader.read_be_u32()? != 24 {
93                return decode_error("common (alac): invalid channel layout info size");
94            }
95
96            // The channel layout info identifier should be the ascii string "chan".
97            if reader.read_quad_bytes()? != *b"chan" {
98                return decode_error("common (alac): invalid channel layout info id");
99            }
100
101            // The channel layout info version must be 0.
102            if reader.read_be_u32()? != 0 {
103                return decode_error("common (alac): invalid channel layout info version");
104            }
105
106            // Read the channel layout tag. The numerical value of this tag is defined by the Apple
107            // CoreAudio API.
108            let layout_channels = match reader.read_be_u32()? {
109                // 100 << 16
110                0x64_0001 => layouts::CHANNEL_LAYOUT_MONO,
111                // 101 << 16
112                0x65_0002 => layouts::CHANNEL_LAYOUT_STEREO,
113                // 113 << 16
114                0x71_0003 => layouts::CHANNEL_LAYOUT_MPEG_3P0_B,
115                // 116 << 16
116                0x74_0004 => layouts::CHANNEL_LAYOUT_MPEG_4P0_B,
117                // 120 << 16
118                0x78_0005 => layouts::CHANNEL_LAYOUT_MPEG_5P0_D,
119                // 124 << 16
120                0x7c_0006 => layouts::CHANNEL_LAYOUT_MPEG_5P1_D,
121                // 142 << 16
122                0x8e_0007 => layouts::CHANNEL_LAYOUT_AAC_6P1,
123                // 127 << 16
124                0x7f_0008 => layouts::CHANNEL_LAYOUT_MPEG_7P1_B,
125                _ => return decode_error("common (alac): invalid channel layout tag"),
126            };
127
128            // The number of channels stated in the mandatory part of the magic cookie should match
129            // the number of channels implicit to the channel layout.
130            if config.num_channels != layout_channels.count() as u8 {
131                return decode_error(
132                    "common (alac): the number of channels differs from the channel layout",
133                );
134            }
135
136            // The next two fields are reserved and should be 0.
137            if reader.read_be_u32()? != 0 || reader.read_be_u32()? != 0 {
138                return decode_error(
139                    "common (alac): reserved values in channel layout info are not 0",
140                );
141            }
142
143            layout_channels
144        }
145        else {
146            // If extra channel information is not provided, use the number of channels to assign
147            // a channel layout.
148            //
149            // TODO: If the number of channels is > 2, then the additional channels are considered
150            // discrete and not part of a channel layout. However, Symphonia does not support
151            // discrete/auxiliary channels so the standard ALAC channel layouts are used for now.
152            match config.num_channels {
153                1 => layouts::CHANNEL_LAYOUT_MONO,
154                2 => layouts::CHANNEL_LAYOUT_STEREO,
155                3 => layouts::CHANNEL_LAYOUT_MPEG_3P0_B,
156                4 => layouts::CHANNEL_LAYOUT_MPEG_4P0_B,
157                5 => layouts::CHANNEL_LAYOUT_MPEG_5P0_D,
158                6 => layouts::CHANNEL_LAYOUT_MPEG_5P1_D,
159                7 => layouts::CHANNEL_LAYOUT_AAC_6P1,
160                8 => layouts::CHANNEL_LAYOUT_MPEG_7P1_B,
161                _ => {
162                    return decode_error(
163                        "common (alac): unknown channel layout for number of channels",
164                    );
165                }
166            }
167        };
168
169        Ok(config)
170    }
171}
