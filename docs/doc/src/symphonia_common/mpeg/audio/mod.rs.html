164/// MPEG4 audio sample rate result.
165pub enum Mpeg4AudioSampleRate {
166    /// The sample rate is known.
167    SampleRate(u32),
168    /// The sample rate should be read manually.
169    ///
170    /// For the MPEG4 Audio Specific Config, the sample rate should be read as the next 24-bits
171    /// after the sample rate index. For ADTS, this is an error.
172    Escape,
173    /// The sample rate index was invalid.
174    Invalid,
175}
176
177/// Try to get the audio sample rate given the sample rate index.
178pub fn get_mpeg4_audio_sample_rate_by_index(index: u32) -> Mpeg4AudioSampleRate {
179    const MPEG4_AUDIO_SAMPLE_RATES: [u32; 13] =
180        [96000, 88200, 64000, 48000, 44100, 32000, 24000, 22050, 16000, 12000, 11025, 8000, 7350];
181
182    match index {
183        0..=12 => Mpeg4AudioSampleRate::SampleRate(MPEG4_AUDIO_SAMPLE_RATES[index as usize]),
184        15 => Mpeg4AudioSampleRate::Escape,
185        _ => Mpeg4AudioSampleRate::Invalid,
186    }
187}

189/// MPEG4 audio channel configuration result.
190pub enum Mpeg4AudioChannels {
191    /// The channel configuration is known.
192    Channels(Channels),
193    /// The channel configuration is defined in-band based on the audio object type (e.g., from the
194    /// program configuration element).
195    Escape,
196    /// The channel configuration index was invalid.
197    Invalid,
198}
199
200/// Try to get the set of channels given the configuration index.
201pub fn get_mpeg4_audio_channels_by_config_index(index: u32) -> Mpeg4AudioChannels {
202    let channels = match index {
203        0 => return Mpeg4AudioChannels::Escape,
204        1 => layouts::CHANNEL_LAYOUT_MONO,
205        2 => layouts::CHANNEL_LAYOUT_STEREO,
206        3 => layouts::CHANNEL_LAYOUT_AAC_3P0,
207        4 => layouts::CHANNEL_LAYOUT_AAC_4P0,
208        5 => layouts::CHANNEL_LAYOUT_AAC_5P0,
209        6 => layouts::CHANNEL_LAYOUT_AAC_5P1,
210        7 => layouts::CHANNEL_LAYOUT_AAC_7P1,
211        _ => return Mpeg4AudioChannels::Invalid,
212    };
213    Mpeg4AudioChannels::Channels(channels)
214}

216/// MPEG4 Audio Specific Configuration.
217#[non_exhaustive]
218#[derive(Debug, Default, Clone, PartialEq, Eq)]
219pub struct AudioSpecificConfig {
220    pub object_type: AudioObjectType,
221    pub sample_rate: u32,
222    pub channels: Option<Channels>,
223    pub samples: usize,
224    pub sbr_ps_info: Option<(u32, Option<Channels>)>,
225    pub sbr_present: bool,
226    pub ps_present: bool,
227}
228
229impl AudioSpecificConfig {
230    /// Read the audio specific configuration from the provided buffer. ISO14496-3-2009
231    pub fn read(buf: &[u8]) -> Result<AudioSpecificConfig> {
232        let mut bs = BitReaderLtr::new(buf);
233
234        let mut asc = AudioSpecificConfig {
235            object_type: Self::read_audio_object_type(&mut bs)?,
236            sample_rate: Self::read_sampling_frequency(&mut bs)?,
237            ..Default::default()
238        };
239
240        if asc.sample_rate == 0 {
241            return decode_error("common (mp4a): a sample rate of 0 is invalid");
242        }
243
244        asc.channels = Self::read_channel_config(&mut bs)?;
245
246        if (asc.object_type == AudioObjectType::Sbr) || (asc.object_type == AudioObjectType::Ps) {
247            asc.sbr_present = true;
248            if asc.object_type == AudioObjectType::Ps {
249                asc.ps_present = true;
250            }
251            let ext_srate = Self::read_sampling_frequency(&mut bs)?;
252            asc.object_type = Self::read_audio_object_type(&mut bs)?;
253
254            let ext_chans = if asc.object_type == AudioObjectType::ErBsac {
255                Self::read_channel_config(&mut bs)?
256            }
257            else {
258                None
259            };
260
261            asc.sbr_ps_info = Some((ext_srate, ext_chans));
262        }
263
264        match asc.object_type {
265            AudioObjectType::Main
266            | AudioObjectType::Lc
267            | AudioObjectType::Ssr
268            | AudioObjectType::Scalable
269            | AudioObjectType::TwinVq
270            | AudioObjectType::ErAacLc
271            | AudioObjectType::ErAacLtp
272            | AudioObjectType::ErAacScalable
273            | AudioObjectType::ErTwinVq
274            | AudioObjectType::ErBsac
275            | AudioObjectType::ErAacLd => {
276                // GASpecificConfig
277                let short_frame = bs.read_bool()?;
278
279                asc.samples = if short_frame { 960 } else { 1024 };
280
281                let depends_on_core = bs.read_bool()?;
282
283                if depends_on_core {
284                    let _delay = bs.read_bits_leq32(14)?;
285                }
286
287                let extension_flag = bs.read_bool()?;
288
289                if asc.channels.is_none() {
290                    return unsupported_error("common (mp4a): program config element");
291                }
292
293                if (asc.object_type == AudioObjectType::Scalable)
294                    || (asc.object_type == AudioObjectType::ErAacScalable)
295                {
296                    let _layer = bs.read_bits_leq32(3)?;
297                }
298
299                if extension_flag {
300                    if asc.object_type == AudioObjectType::ErBsac {
301                        let _num_subframes = bs.read_bits_leq32(5)? as usize;
302                        let _layer_length = bs.read_bits_leq32(11)?;
303                    }
304
305                    if (asc.object_type == AudioObjectType::ErAacLc)
306                        || (asc.object_type == AudioObjectType::ErAacLtp)
307                        || (asc.object_type == AudioObjectType::ErAacScalable)
308                        || (asc.object_type == AudioObjectType::ErAacLd)
309                    {
310                        let _section_data_resilience = bs.read_bool()?;
311                        let _scalefactors_resilience = bs.read_bool()?;
312                        let _spectral_data_resilience = bs.read_bool()?;
313                    }
314
315                    let extension_flag3 = bs.read_bool()?;
316
317                    if extension_flag3 {
318                        return unsupported_error("common (mp4a): version3 extensions");
319                    }
320                }
321            }
322            AudioObjectType::Celp => {
323                return unsupported_error("common (mp4a): CELP config");
324            }
325            AudioObjectType::Hvxc => {
326                return unsupported_error("common (mp4a): HVXC config");
327            }
328            AudioObjectType::Ttsi => {
329                return unsupported_error("common (mp4a): TTS config");
330            }
331            AudioObjectType::MainSynth
332            | AudioObjectType::WavetableSynth
333            | AudioObjectType::GeneralMidi
334            | AudioObjectType::Algorithmic => {
335                return unsupported_error("common (mp4a): structured audio config");
336            }
337            AudioObjectType::ErCelp => {
338                return unsupported_error("common (mp4a): ER CELP config");
339            }
340            AudioObjectType::ErHvxc => {
341                return unsupported_error("common (mp4a): ER HVXC config");
342            }
343            AudioObjectType::ErHiln | AudioObjectType::ErParametric => {
344                return unsupported_error("common (mp4a): parametric config");
345            }
346            AudioObjectType::Ssc => {
347                return unsupported_error("common (mp4a): SSC config");
348            }
349            AudioObjectType::MpegSurround => {
350                // bs.ignore_bits(1)?; // sacPayloadEmbedding
351                return unsupported_error("common (mp4a): MPEG Surround config");
352            }
353            AudioObjectType::Layer1 | AudioObjectType::Layer2 | AudioObjectType::Layer3 => {
354                return unsupported_error("common (mp4a): MPEG Layer 1/2/3 config");
355            }
356            AudioObjectType::Dst => {
357                return unsupported_error("common (mp4a): DST config");
358            }
359            AudioObjectType::Als => {
360                // bs.ignore_bits(5)?; // fillBits
361                return unsupported_error("common (mp4a): ALS config");
362            }
363            AudioObjectType::Sls | AudioObjectType::SlsNonCore => {
364                return unsupported_error("common (mp4a): SLS config");
365            }
366            AudioObjectType::ErAacEld => {
367                return unsupported_error("common (mp4a): ELD config");
368            }
369            AudioObjectType::SmrSimple | AudioObjectType::SmrMain => {
370                return unsupported_error("common (mp4a): symbolic music config");
371            }
372            _ => {}
373        };
374
375        match asc.object_type {
376            AudioObjectType::ErAacLc
377            | AudioObjectType::ErAacLtp
378            | AudioObjectType::ErAacScalable
379            | AudioObjectType::ErTwinVq
380            | AudioObjectType::ErBsac
381            | AudioObjectType::ErAacLd
382            | AudioObjectType::ErCelp
383            | AudioObjectType::ErHvxc
384            | AudioObjectType::ErHiln
385            | AudioObjectType::ErParametric
386            | AudioObjectType::ErAacEld => {
387                let ep_config = bs.read_bits_leq32(2)?;
388
389                if (ep_config == 2) || (ep_config == 3) {
390                    return unsupported_error("common (mp4a): error protection config");
391                }
392                // if ep_config == 3 {
393                //     let direct_mapping = bs.read_bit()?;
394                //     validate!(direct_mapping);
395                // }
396            }
397            _ => {}
398        };
399
400        if asc.sbr_ps_info.is_some() && (bs.bits_left() >= 16) {
401            let sync = bs.read_bits_leq32(11)?;
402
403            if sync == 0x2B7 {
404                let ext_otype = Self::read_audio_object_type(&mut bs)?;
405                if ext_otype == AudioObjectType::Sbr {
406                    asc.sbr_present = bs.read_bool()?;
407                    if asc.sbr_present {
408                        let _ext_srate = Self::read_sampling_frequency(&mut bs)?;
409                        if bs.bits_left() >= 12 {
410                            let sync = bs.read_bits_leq32(11)?;
411                            if sync == 0x548 {
412                                asc.ps_present = bs.read_bool()?;
413                            }
414                        }
415                    }
416                }
417                if ext_otype == AudioObjectType::Ps {
418                    asc.sbr_present = bs.read_bool()?;
419                    if asc.sbr_present {
420                        let _ext_srate = Self::read_sampling_frequency(&mut bs)?;
421                    }
422                    let _ext_channels = bs.read_bits_leq32(4)?;
423                }
424            }
425        }
426
427        Ok(asc)
428    }
429
430    fn read_audio_object_type<B: ReadBitsLtr>(bs: &mut B) -> Result<AudioObjectType> {
431        let index = match bs.read_bits_leq32(5)? {
432            index if index < 31 => index as usize,
433            31 => (bs.read_bits_leq32(6)? + 32) as usize,
434            _ => unreachable!(),
435        };
436
437        let aot = AUDIO_OBJECT_TYPES.get(index).copied().unwrap_or(AudioObjectType::Unknown);
438
439        Ok(aot)
440    }
441
442    fn read_sampling_frequency<B: ReadBitsLtr>(bs: &mut B) -> Result<u32> {
443        let index = bs.read_bits_leq32(4)?;
444        let rate = match get_mpeg4_audio_sample_rate_by_index(index) {
445            Mpeg4AudioSampleRate::SampleRate(rate) => rate,
446            Mpeg4AudioSampleRate::Escape => bs.read_bits_leq32(24)?,
447            Mpeg4AudioSampleRate::Invalid => {
448                return decode_error("common (mp4a): invalid sample rate");
449            }
450        };
451        Ok(rate)
452    }
453
454    fn read_channel_config<B: ReadBitsLtr>(bs: &mut B) -> Result<Option<Channels>> {
455        let index = bs.read_bits_leq32(4)?;
456        let channels = match get_mpeg4_audio_channels_by_config_index(index) {
457            Mpeg4AudioChannels::Channels(channels) => Some(channels),
458            Mpeg4AudioChannels::Escape => None,
459            Mpeg4AudioChannels::Invalid => {
460                return decode_error("common (mp4a): invalid channel configuration");
461            }
462        };
463        Ok(channels)
464    }
465}

467pub fn get_audio_codec_profile(asc: &AudioSpecificConfig) -> Option<CodecProfile> {
468    match asc.object_type {
469        AudioObjectType::Main => Some(CODEC_PROFILE_AAC_MAIN),
470        AudioObjectType::Ssr => Some(CODEC_PROFILE_AAC_SSR),
471        AudioObjectType::Ltp => Some(CODEC_PROFILE_AAC_LTP),
472        AudioObjectType::Lc => {
473            if asc.ps_present {
474                Some(CODEC_PROFILE_AAC_HE_V2)
475            }
476            else if asc.sbr_present {
477                Some(CODEC_PROFILE_AAC_HE)
478            }
479            else {
480                Some(CODEC_PROFILE_AAC_LC)
481            }
482        }
483        _ => None,
484    }
