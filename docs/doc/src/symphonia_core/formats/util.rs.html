8//! Helper utilities for implementing media demuxers.
9
10use crate::units::Timestamp;
11
12/// A `SeekPoint` is a mapping between a sample or frame number to byte offset within a media
13/// stream.
14#[derive(Copy, Clone, Debug, PartialEq, Eq)]
15pub struct SeekPoint {
16    /// The frame or sample timestamp of the `SeekPoint`.
17    pub frame_ts: Timestamp,
18    /// The byte offset of the `SeekPoint`s timestamp relative to a format-specific location.
19    pub byte_offset: u64,
20    /// The number of frames the `SeekPoint` covers.
21    pub n_frames: u32,
22}
23
24impl SeekPoint {
25    fn new(frame_ts: Timestamp, byte_offset: u64, n_frames: u32) -> Self {
26        SeekPoint { frame_ts, byte_offset, n_frames }
27    }
28}
29
30/// A `SeekIndex` stores `SeekPoint`s (generally a sample or frame number to byte offset) within
31/// a media stream and provides methods to efficiently search for the nearest `SeekPoint`(s)
32/// given a timestamp.
33///
34/// A `SeekIndex` does not require complete coverage of the entire media stream. However, the
35/// better the coverage, the smaller the manual search range the `SeekIndex` will return.
36#[derive(Default)]
37pub struct SeekIndex {
38    points: Vec<SeekPoint>,
39}
40
41/// `SeekSearchResult` is the return value for a search on a `SeekIndex`. It returns a range of
42/// `SeekPoint`s a `FormatReader` should search to find the desired timestamp. Ranges are
43/// lower-bound inclusive, and upper-bound exclusive.
44#[derive(Copy, Clone, Debug, PartialEq, Eq)]
45pub enum SeekSearchResult {
46    /// The `SeekIndex` is empty so the desired timestamp could not be found. The entire stream
47    /// should be searched for the desired timestamp.
48    Stream,
49    /// The desired timestamp can be found before, the `SeekPoint`. The stream should be
50    /// searched for the desired timestamp from the start of the stream up-to, but not
51    /// including, the `SeekPoint`.
52    Upper(SeekPoint),
53    /// The desired timestamp can be found at, or after, the `SeekPoint`. The stream should be
54    /// searched for the desired timestamp starting at the provided `SeekPoint` up-to the end of
55    /// the stream.
56    Lower(SeekPoint),
57    /// The desired timestamp can be found within the range. The stream should be searched for
58    /// the desired starting at the first `SeekPoint` up-to, but not-including, the second
59    /// `SeekPoint`.
60    Range(SeekPoint, SeekPoint),
61}
62
63impl SeekIndex {
64    /// Create an empty `SeekIndex`
65    pub fn new() -> SeekIndex {
66        SeekIndex { points: Vec::new() }
67    }
68
69    /// Insert a `SeekPoint` into the index.
70    pub fn insert(&mut self, ts: Timestamp, byte_offset: u64, n_frames: u32) {
71        // Create the seek point.
72        let seek_point = SeekPoint::new(ts, byte_offset, n_frames);
73
74        // Get the timestamp of the last entry in the index.
75        let (last_ts, last_offset) =
76            self.points.last().map_or((Timestamp::MIN, 0), |p| (p.frame_ts, p.byte_offset));
77
78        // If the seek point has a timestamp greater-than and byte offset greater-than or equal to
79        // the last entry in the index, then simply append it to the index.
80        if ts > last_ts && byte_offset >= last_offset {
81            self.points.push(seek_point)
82        }
83        else if ts < last_ts {
84            // If the seek point has a timestamp less-than the last entry in the index, then the
85            // insertion point must be found. This case should rarely occur.
86            let i =
87                self.points.partition_point(|p| ts > p.frame_ts && byte_offset >= p.byte_offset);
88
89            // Insert if the point found or if the points are empty
90            if i < self.points.len() || i == 0 {
91                self.points.insert(i, seek_point);
92            }
93        }
94    }
95
96    /// Search the index to find a bounded range of bytes wherein the specified frame timestamp
97    /// will be contained. If the index is empty, this function simply returns a result
98    /// indicating the entire stream should be searched manually.
99    pub fn search(&self, frame_ts: Timestamp) -> SeekSearchResult {
100        // The index must contain atleast one SeekPoint to return a useful result.
101        if !self.points.is_empty() {
102            let mut lower = 0;
103            let mut upper = self.points.len() - 1;
104
105            // If the desired timestamp is less than the first SeekPoint within the index,
106            // indicate that the stream should be searched from the beginning.
107            if frame_ts < self.points[lower].frame_ts {
108                return SeekSearchResult::Upper(self.points[lower]);
109            }
110            // If the desired timestamp is greater than or equal to the last SeekPoint within
111            // the index, indicate that the stream should be searched from the last SeekPoint.
112            else if frame_ts >= self.points[upper].frame_ts {
113                return SeekSearchResult::Lower(self.points[upper]);
114            }
115
116            // Desired timestamp is between the lower and upper indicies. Perform a binary
117            // search to find a range of SeekPoints containing the desired timestamp. The binary
118            // search exits when either two adjacent SeekPoints or a single SeekPoint is found.
119            while upper - lower > 1 {
120                let mid = (lower + upper) / 2;
121                let mid_ts = self.points[mid].frame_ts;
122
123                if frame_ts < mid_ts {
124                    upper = mid;
125                }
126                else {
127                    lower = mid;
128                }
129            }
130
131            return SeekSearchResult::Range(self.points[lower], self.points[upper]);
132        }
133
134        // The index is empty, the stream must be searched manually.
135        SeekSearchResult::Stream
136    }
137}

139#[cfg(test)]
140mod tests {
141    use crate::units::Timestamp;
142
143    use super::{SeekIndex, SeekPoint, SeekSearchResult};
144
145    #[test]
146    fn verify_seek_index_search() {
147        let mut index = SeekIndex::new();
148        // Normal index insert
149        index.insert(Timestamp::new(479232), 706812, 1152);
150        index.insert(Timestamp::new(959616), 1421536, 1152);
151        index.insert(Timestamp::new(1919232), 2833241, 1152);
152        index.insert(Timestamp::new(2399616), 3546987, 1152);
153        index.insert(Timestamp::new(2880000), 4259455, 1152);
154
155        // Search for point lower than the first entry
156        assert_eq!(
157            index.search(Timestamp::new(0)),
158            SeekSearchResult::Upper(SeekPoint::new(Timestamp::new(479232), 706812, 1152))
159        );
160
161        // Search for point higher than last entry
162        assert_eq!(
163            index.search(Timestamp::new(3000000)),
164            SeekSearchResult::Lower(SeekPoint::new(Timestamp::new(2880000), 4259455, 1152))
165        );
166
167        // Search for point that has equal timestamp with some index
168        assert_eq!(
169            index.search(Timestamp::new(959616)),
170            SeekSearchResult::Range(
171                SeekPoint::new(Timestamp::new(959616), 1421536, 1152),
172                SeekPoint::new(Timestamp::new(1919232), 2833241, 1152)
173            )
174        );
175
176        // Index insert out of order
177        index.insert(Timestamp::new(1440000), 2132419, 1152);
178        index.insert(Timestamp::new(-78000), 20, 1152);
179        index.insert(Timestamp::new(-50000), 45000, 1152);
180
181        // Search for a negative timestamp.
182        assert_eq!(
183            index.search(Timestamp::MIN),
184            SeekSearchResult::Upper(SeekPoint::new(Timestamp::new(-78000), 20, 1152))
185        );
186
187        assert_eq!(
188            index.search(Timestamp::new(-69000)),
189            SeekSearchResult::Range(
190                SeekPoint::new(Timestamp::new(-78000), 20, 1152),
191                SeekPoint::new(Timestamp::new(-50000), 45000, 1152)
192            )
193        );
194
195        // Search for 0 again.
196        assert_eq!(
197            index.search(Timestamp::new(0)),
198            SeekSearchResult::Range(
199                SeekPoint::new(Timestamp::new(-50000), 45000, 1152),
200                SeekPoint::new(Timestamp::new(479232), 706812, 1152)
201            )
202        );
203
204        // Search for point that have out of order index when inserting
205        assert_eq!(
206            index.search(Timestamp::new(1000000)),
207            SeekSearchResult::Range(
208                SeekPoint::new(Timestamp::new(959616), 1421536, 1152),
209                SeekPoint::new(Timestamp::new(1440000), 2132419, 1152)
210            )
211        );
212
213        // Index insert with byte_offset less than last entry
214        index.insert(Timestamp::new(3359232), 0, 0);
215
216        // Search for ignored point because byte_offset less than last entry
217        assert_eq!(
218            index.search(Timestamp::new(3359232)),
219            SeekSearchResult::Lower(SeekPoint::new(Timestamp::new(2880000), 4259455, 1152))
220        );
221    }
