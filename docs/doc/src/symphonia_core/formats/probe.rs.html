23    fn fnv1a32(value: &[u8; 2]) -> u32 {
24        const INIT: u32 = 0x811c_9dc5;
25        const PRIME: u32 = 0x0100_0193;
26
27        let mut state = INIT;
28
29        for byte in value.iter() {
30            state = (state ^ u32::from(*byte)).wrapping_mul(PRIME);
31        }
32
33        state
34    }
35
36    pub struct BloomFilter {
37        filter: Box<[u64]>,
38    }
39
40    impl Default for BloomFilter {
41        fn default() -> Self {
42            BloomFilter { filter: vec![0; BloomFilter::M >> 6].into_boxed_slice() }
43        }
44    }
45
46    impl BloomFilter {
47        /// The number of bits, m, used by the bloom filter. Use 16384 bits (2KiB) by default.
48        const M: usize = 2 * 1024 * 8;
49
50        pub fn insert(&mut self, key: &[u8; 2]) {
51            let hash = fnv1a32(key);
52
53            let h0 = (hash >> 16) as u16;
54            let h1 = (hash >> 0) as u16;
55
56            let i0 = h0 as usize & (BloomFilter::M - 1);
57            let i1 = h0.wrapping_add(h1.wrapping_mul(1)) as usize & (BloomFilter::M - 1);
58            let i2 = h0.wrapping_add(h1.wrapping_mul(2)) as usize & (BloomFilter::M - 1);
59
60            self.filter[i0 >> 6] |= 1 << (i0 & 63);
61            self.filter[i1 >> 6] |= 1 << (i1 & 63);
62            self.filter[i2 >> 6] |= 1 << (i2 & 63);
63        }
64
65        pub fn may_contain(&self, key: &[u8; 2]) -> bool {
66            let hash = fnv1a32(key);
67
68            let h0 = (hash >> 16) as u16;
69            let h1 = (hash >> 0) as u16;
70
71            let i0 = h0 as usize & (BloomFilter::M - 1);
72            let i1 = h0.wrapping_add(h1.wrapping_mul(1)) as usize & (BloomFilter::M - 1);
73            let i2 = h0.wrapping_add(h1.wrapping_mul(2)) as usize & (BloomFilter::M - 1);
74
75            if (self.filter[i0 >> 6] & (1 << (i0 & 63))) == 0 {
76                return false;
77            }
78            if (self.filter[i1 >> 6] & (1 << (i1 & 63))) == 0 {
79                return false;
80            }
81            if (self.filter[i2 >> 6] & (1 << (i2 & 63))) == 0 {
82                return false;
83            }
84
85            true
86        }
87    }

90/// Anchors are explicit fixed offsets from the end of a media stream that should be searched for
91/// format markers.
92#[derive(Copy, Clone)]
93pub enum Anchors {
94    /// No anchors.
95    None,
96    /// Exclusive anchors indicate that the format markers will only searched for at the anchor
97    /// points.
98    Exclusive(&'static [u32]),
99    /// Supplemental anchors indicate that the format markers will first be scanned for at the
100    /// anchor points, then from the beginning of the stream.
101    Supplemental(&'static [u32]),

148/// A probe match is the result of one probe iteration on a given media source stream.
149///
150/// A probe match provides a basic description of the matching container or metadata format, and a
151/// factory function to instantiate a reader for it.
152#[derive(Copy, Clone)]
153enum ProbeMatch {
154    Format {
155        /// A basic description about the container format.
156        info: FormatInfo,
157        /// A factory function to create an instance of the matching format reader.
158        factory: FormatFactoryFn,
159    },
160    Metadata {
161        /// A basic description about the metadata format.
162        info: MetadataInfo,
163        /// A factory function to create an instance of the matching metadata reader.
164        factory: MetadataFactoryFn,
165    },
166}

168/// A function pointer to the score function of the registered probeable.
169type ScoreFn = fn(ScopedStream<&mut MediaSourceStream<'_>>) -> Result<Score>;
170
171/// Private/internal generalized representation of a probeable format or metadata reader.
172#[derive(Copy, Clone)]
173struct GenericProbeMatch {
174    /// The match specification.
175    spec: ProbeDataMatchSpec,
176    /// A function to assign a likelyhood score that the media source, readable with scoped access
177    /// via. the provided stream, is the start of a metadate or container format
178    score: ScoreFn,
179    /// Format or metadata reader specific match information.
180    specific: ProbeMatch,
181    /// Anchors.
182    anchors: Anchors,
183}
184
185impl GenericProbeMatch {
186    /// Returns `true` if this match should be tested based on the probe type.
187    fn should_test(&self, is_trailing_probe: bool) -> bool {
188        match self.anchors {
189            // Matches with no anchors should only be tested on non-trailing probes.
190            Anchors::None => !is_trailing_probe,
191            // Matches with exclusive anchors should only be tested on trailing probes.
192            Anchors::Exclusive(_) => is_trailing_probe,
193            // Matches with supplemental anchors should always be tested.
194            Anchors::Supplemental(_) => true,
195        }
196    }
197}

199/// The result of a scoring operation.
200pub enum Score {
201    /// The format is not supported.
202    Unsupported,
203    /// The format is supported with a confidence between 0 (not confident) and 255 (very
204    /// confident).
205    Supported(u8),
206}
207
208/// The `Scoreable` trait defines the scoring functionality a reader must implement to support
209/// probing for a container or metadata format.
210pub trait Scoreable {
211    /// Using scoped access to a `MediaSourceStream`, calculate and return a value between 0 and 255
212    /// indicating the confidence of the reader in decoding or parsing the stream.
213    ///
214    /// If the format is definitely not supported, then score should return [`Score::Unsupported`]
215    /// since a score of 0 is still considered supported, even if unlikely.
216    ///
217    /// If an error is returned, errors other than [`Error::IoError`] (excluding the unexpected EOF
218    /// kind) are treated as if [`Score::Unsupported`] was returned. All other IO errors abort
219    /// the probe operation.
220    fn score(src: ScopedStream<&mut MediaSourceStream<'_>>) -> Result<Score>;
221}

253/// A `Hint` provides additional information and context when probing a media source stream.
254///
255/// For example, the `Probe` cannot examine the extension or mime-type of the media because

285/// Options for controlling the behaviour of a `Probe`.
286#[derive(Copy, Clone, Debug, PartialEq, Eq)]
287pub struct ProbeOptions {
288    /// The maximum number of bytes that will be scanned from the media source before giving up.
289    ///
290    /// The default is 1 MB, the maximum is 4 GB.
291    pub max_probe_depth: u32,
292    /// The maximum number of bytes a score operation may read before it will be forced to abort.
293    ///
294    /// The larger this value is, the larger the media source buffer must be, and therefore the more
295    /// memory is consumed.
296    ///
297    /// The default is 16 kB, the maximum is 64 kB.
298    pub max_score_depth: u16,
299}
300
301impl Default for ProbeOptions {
302    fn default() -> Self {
303        Self {
304            max_probe_depth: 1 * 1024 * 1024, // 1 MB
305            max_score_depth: 16 * 1024,       // 16 kB
306        }
307    }
308}

310/// `Probe` scans a `MediaSourceStream` for metadata and container formats, and provides an
311/// iterator-like interface to instantiate readers for the formats encountered.
312#[derive(Default)]
313pub struct Probe {
314    filter: bloom::BloomFilter,
315    preferred: Vec<GenericProbeMatch>,
316    standard: Vec<GenericProbeMatch>,
317    fallback: Vec<GenericProbeMatch>,
318    anchors: Vec<u32>,
319    opts: ProbeOptions,
320}
321
322impl Probe {
323    /// Instantiate a probe with default options.
324    pub fn new() -> Self {
325        Probe::new_with_options(&Default::default())
326    }

328    /// Instantiate a probe with custom options.
329    pub fn new_with_options(opts: &ProbeOptions) -> Self {
330        Probe { opts: *opts, ..Default::default() }

333    /// Register the parameterized format reader at the standard tier.
334    pub fn register_format<P>(&mut self)
335    where
336        for<'a> P: ProbeableFormat<'a>,
337    {
338        self.register_format_at_tier::<P>(Tier::Standard);
339    }
340
341    /// Register the parameterized metadata reader at the standard tier.
342    pub fn register_metadata<P>(&mut self)
343    where
344        for<'a> P: ProbeableMetadata<'a>,
345    {
346        self.register_metadata_at_tier::<P>(Tier::Standard);
347    }
348
349    /// Register the parameterized format reader at a specific tier.
350    pub fn register_format_at_tier<P>(&mut self, tier: Tier)
351    where
352        for<'a> P: ProbeableFormat<'a>,
353    {
354        for data in P::probe_data() {
355            // Build a generic format probe candidate.
356            let candidate = GenericProbeMatch {
357                spec: data.spec,
358                score: P::score,
359                specific: ProbeMatch::Format {
360                    info: data.info,
361                    factory: |mss, opts| P::try_probe_new(mss, opts),
362                },
363                anchors: Anchors::None,
364            };
365
366            self.register(tier, candidate);
367        }
368    }
369
370    /// Register the parameterized metadata reader at a specific tier.
371    pub fn register_metadata_at_tier<P>(&mut self, tier: Tier)
372    where
373        for<'a> P: ProbeableMetadata<'a>,
374    {
375        for data in P::probe_data() {
376            // Build a generic metadata probe candidate.
377            let candidate = GenericProbeMatch {
378                spec: data.spec,
379                score: P::score,
380                specific: ProbeMatch::Metadata {
381                    info: data.info,
382                    factory: |mss, opts| P::try_probe_new(mss, opts),
383                },
384                anchors: data.anchors,
385            };
386
387            match data.anchors {
388                Anchors::Exclusive(anchors) | Anchors::Supplemental(anchors) => {
389                    self.anchors.extend_from_slice(anchors);
390                    // Iterating a stream backwards is very inefficient because a seek backwards
391                    // invalidates the entire lookahead buffer. Therefore, the list of anchors is
392                    // sorted from largest to smallest offset so that the iteration can be in the
393                    // forward direction.
394                    self.anchors.sort_by(|a, b| b.cmp(a));
395                    self.anchors.dedup();
396                }
397                _ => (),
398            }
399
400            // Register candidate.
401            self.register(tier, candidate);
402        }
403    }
404
405    fn register(&mut self, tier: Tier, candidate: GenericProbeMatch) {
406        // Insert 2-byte prefixes for each marker into the bloom filter.
407        for marker in candidate.spec.markers {
408            let mut prefix = [0u8; 2];
409
410            match marker.len() {
411                2..=16 => prefix.copy_from_slice(&marker[0..2]),
412                _ => panic!("invalid marker length (only 2-16 bytes supported)."),
413            }
414
415            self.filter.insert(&prefix);
416        }
417
418        // Register at the desired tier.
419        match tier {
420            Tier::Preferred => self.preferred.push(candidate),
421            Tier::Standard => self.standard.push(candidate),
422            Tier::Fallback => self.fallback.push(candidate),
423        }
424    }

426    /// Searches the provided `MediaSourceStream` for a container format. Any metadata that is read
427    /// during the search will be queued and attached to the `FormatReader` instance once a
428    /// container format is found.
429    pub fn probe<'s>(
430        &self,
431        hint: &Hint,
432        mut mss: MediaSourceStream<'s>,
433        mut fmt_opts: FormatOptions,
434        meta_opts: MetadataOptions,
435    ) -> Result<Box<dyn FormatReader + 's>> {
436        // Probe for trailing metadata only if the media source stream is seekable, and the length
437        // is known.
438        if mss.is_seekable() {
439            if let Some(end) = mss.byte_len() {
440                // The initial position of the stream will be restored after probing for trailing
441                // metadata.
442                let init_pos = mss.pos();
443
444                mss = self.probe_trailing(mss, end, &mut fmt_opts, meta_opts)?;
445
446                // Restore position.
447                mss.seek(SeekFrom::Start(init_pos))?;
448            }
449        }
450
451        // Loop over all elements in the stream until a container format is found.
452        loop {
453            match self.next(&mut mss, hint)? {
454                // If a container format is found, return an instance to it's reader.
455                ProbeMatch::Format { factory, .. } => {
456                    // Instantiate the format reader.
457                    return factory(mss, fmt_opts);
458                }
459                // If metadata was found, instantiate the metadata reader, read the metadata, and
460                // push it onto the metadata log.
461                ProbeMatch::Metadata { factory, .. } => {
462                    let mut reader = factory(mss, meta_opts)?;
463                    read_and_append_metadata(&mut reader, &mut fmt_opts)?;
464                    mss = reader.into_inner();
465                }
466            }
467        }
468
469        // This function returns when either the end-of-stream is reached, an error occurs, or a
470        // container format is found.
471    }

473    /// Scans for trailing metadata relative to the end position of the provided
474    /// `MediaSourceStream`.
475    fn probe_trailing<'s>(
476        &self,
477        mut mss: MediaSourceStream<'s>,
478        end: u64,
479        fmt_opts: &mut FormatOptions,
480        meta_opts: MetadataOptions,
481    ) -> Result<MediaSourceStream<'s>> {
482        debug_assert!(mss.is_seekable());
483
484        // The position of the media source stream after a reader successfully reads the stream.
485        let mut last_reader_end = 0;
486
487        // Probe the anchor points.
488        for anchor in self.anchors.iter().copied().map(u64::from) {
489            // Skip anchor if the anchor offset exceeds the source length, or intersects the last
490            // read preceeds the end of the last successful read.
491            if anchor > end || end - anchor < last_reader_end {
492                debug!("skipping trailing metadata offset -{anchor}");
493                continue;
494            }
495
496            trace!("probing for trailing metadata at offset -{} ({})", anchor, end - anchor);
497
498            // Seek to the anchor point. In this best case, this seek will only invalidate the
499            // lookahead buffer once for the first anchor, and then seek around the lookahead buffer
500            // for the remainder.
501            let anchor_pos = end - anchor;
502
503            if mss.seek_buffered(anchor_pos) != anchor_pos {
504                mss.seek(SeekFrom::Start(anchor_pos))?;
505            }
506
507            // There is potentially nothing of interest at this offset. Read a 2-byte window to test
508            // with the bloom filter.
509            let win = mss.read_double_bytes()?;
510
511            if self.filter.may_contain(&win) {
512                // Re-align to the start of the marker.
513                mss.seek_buffered_rel(-2);
514
515                if let Some(ProbeMatch::Metadata { factory, .. }) =
516                    self.find_best_reader(&mut mss, true)?
517                {
518                    let mut reader = factory(mss, meta_opts)?;
519
520                    // When reading anchored metadata fails it is reasonable to just ignore the
521                    // error and continue because the MSS will be repositioned immediately after
522                    // anyways.
523                    let is_ok = read_and_append_metadata(&mut reader, fmt_opts).is_ok();
524
525                    if !is_ok {
526                        warn!(
527                            "skipping trailing '{}' metdata due to error while reading it",
528                            reader.metadata_info().short_name
529                        );
530                    }
531
532                    mss = reader.into_inner();
533
534                    // If metadata was successfully read, then only anchor positions that are after
535                    // the reader's current position need to be checked.
536                    if is_ok {
537                        last_reader_end = mss.pos();
538                    }
539                }
540            }
541        }
542
543        Ok(mss)
544    }

546    /// Scans the provided `MediaSourceStream` from the current position for the best next metadata
547    /// or format reader. If a match is found, returns it.
548    fn next(&self, mss: &mut MediaSourceStream<'_>, _hint: &Hint) -> Result<ProbeMatch> {
549        let mut win = 0u16;
550
551        let init_pos = mss.pos();
552        let mut count = 0;
553
554        // Scan the stream byte-by-byte. Shifting each byte through a 2-byte window.
555        while let Ok(byte) = mss.read_byte() {
556            win = (win << 8) | u16::from(byte);
557
558            count += 1;
559
560            if count > self.opts.max_probe_depth {
561                break;
562            }
563
564            if count % 4096 == 0 {
565                debug!(
566                    "searching for format marker... {}+{} / {} bytes",
567                    init_pos, count, self.opts.max_probe_depth
568                );
569            }
570
571            // Use the bloom filter to check if the the 2-byte window may be a prefix of a
572            // registered marker.
573            if self.filter.may_contain(&win.to_be_bytes()) {
574                // Re-align the stream to the start of the marker.
575                mss.seek_buffered_rel(-2);
576
577                // Try to find the best matching format or metadata.
578                if let Some(probed) = self.find_best_reader(mss, false)? {
579                    warn_junk_bytes(mss.pos(), init_pos);
580                    return Ok(probed);
581                }
582
583                // If no match was found, re-align the stream to the end of the 2-byte window, and
584                // continue probing.
585                mss.seek_buffered_rel(2);
586            }
587        }
588
589        if count < self.opts.max_probe_depth {
590            error!("probe reached EOF at {count} bytes");
591        }
592        else {
593            // Could not find any marker within the probe limit.
594            error!("reached probe limit of {} bytes", self.opts.max_probe_depth);
595        }
596
597        unsupported_error("core (probe): no suitable format reader found")
598    }

600    /// Find the best reader.
601    fn find_best_reader(
602        &self,
603        mss: &mut MediaSourceStream,
604        is_trailing: bool,
605    ) -> Result<Option<ProbeMatch>> {
606        // Read upto a 16 byte window starting with the marker.
607        let mut win = [0u8; 16];
608        let win_len = mss.read_buf(&mut win)?;
609
610        // Re-align the stream to the start of the marker for scoring.
611        mss.seek_buffered_rel(-(win_len as isize));
612
613        // TODO: Only pass &win[..win_len].
614
615        // Try to find a descriptor in the preferred tier.
616        if let Some(inst) =
617            find_reader(mss, &self.preferred, win, self.opts.max_score_depth, is_trailing)?
618        {
619            return Ok(Some(inst));
620        }
621
622        // Try to find a descriptor in the standard tier.
623        if let Some(inst) =
624            find_reader(mss, &self.standard, win, self.opts.max_score_depth, is_trailing)?
625        {
626            return Ok(Some(inst));
627        }
628
629        // Try to find a descriptor in the fallback tier.
630        if let Some(inst) =
631            find_reader(mss, &self.fallback, win, self.opts.max_score_depth, is_trailing)?
632        {
633            return Ok(Some(inst));
634        }
635
636        Ok(None)
637    }

640fn read_and_append_metadata<'s>(
641    reader: &mut Box<dyn MetadataReader + 's>,
642    fmt_opts: &mut FormatOptions,
643) -> Result<()> {
644    // Read all metadata and get a metdata revision.
645    let metadata = reader.read_all()?;
646
647    debug!("appending '{}' metadata", reader.metadata_info().short_name);
648
649    // Append it to the metadata log.
650    fmt_opts.external_data.metadata.get_or_insert_with(Default::default).push(metadata.revision);
651
652    // Append relevant side data.
653    for side_data in metadata.side_data {
654        match side_data {
655            MetadataSideData::Chapters(chapters) => {
656                debug!("appending '{}' chapters", reader.metadata_info().short_name);
657
658                fmt_opts.external_data.chapters = Some(chapters)
659            }
660        }
661    }
662
663    // Consume the metadata reader and return the media source stream to the caller.
664    Ok(())
665}

669    descs: &[GenericProbeMatch],
670    win: [u8; 16],
671    max_depth: u16,
672    is_trailing: bool,
673) -> Result<Option<ProbeMatch>> {
674    // Ensure the seekback buffer can satisfy the maximum amount of bytes a score operation may
675    // consume.
676    mss.ensure_seekback_buffer(usize::from(max_depth));
677
678    for desc in descs.iter().filter(|d| d.should_test(is_trailing)) {
679        // If any format descriptor marker matches, then the format should be scored.
680        let should_score = desc.spec.markers.iter().any(|marker| {
681            let is_match = win[0..marker.len()] == **marker;
682
683            if is_match {
684                trace!("found the marker {:x?} @ {} bytes", &win[0..marker.len()], mss.pos());
685            }
686
687            is_match
688        });
689
690        // If a match is found, then score using the descriptor's score function.
691        if should_score {
692            // If supported, return the instantiate.
693            if let Score::Supported(score) = score(desc, mss, max_depth)? {
694                match &desc.specific {
695                    ProbeMatch::Format { info, .. } => {
696                        debug!("selected format reader '{}' with score {}", info.short_name, score)
697                    }
698                    ProbeMatch::Metadata { info, .. } => {
699                        debug!(
700                            "selected metadata reader '{}' with score {}",
701                            info.short_name, score
702                        )
703                    }
704                }
705
706                return Ok(Some(desc.specific));
707            }
708
709            match &desc.specific {
710                ProbeMatch::Format { info, .. } => {
711                    trace!("format reader '{}' failed scoring", info.short_name)
712                }
713                ProbeMatch::Metadata { info, .. } => {
714                    trace!("metadata reader '{}' failed scoring", info.short_name)
715                }
716            }
717        }
718    }
719
720    Ok(None)
721}
722
723fn score(
724    candidate: &GenericProbeMatch,
725    mss: &mut MediaSourceStream,
726    max_depth: u16,
727) -> Result<Score> {
728    // Save the initial position to rewind back to after scoring is complete.
729    let init_pos = mss.pos();
730
731    // Perform the scoring operation.
732    let result = match (candidate.score)(ScopedStream::new(mss, u64::from(max_depth))) {
733        Err(Error::IoError(err)) if err.kind() != std::io::ErrorKind::UnexpectedEof => {
734            // IO errors that are not an unexpected end-of-file (or out-of-bounds) error, abort the
735            // entire probe operation.
736            Err(Error::IoError(err))
737        }
738        Err(_) => {
739            // All other errors are caught and return unsupported.
740            Ok(Score::Unsupported)
741        }
742        result => result,
743    };
744
745    // Rewind to the initial position.
746    mss.seek_buffered(init_pos);
747
748    result
749}
750
751fn warn_junk_bytes(pos: u64, init_pos: u64) {
752    // Warn if junk bytes were skipped.
753    if pos > init_pos {
754        warn!("skipped {} bytes of junk at {}", pos - init_pos, init_pos);
755    }
756}
