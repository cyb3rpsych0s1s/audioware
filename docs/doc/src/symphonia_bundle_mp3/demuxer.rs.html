39/// `MpaReader` implements a demuxer for the MPEG1 and MPEG2 audio elementary stream.

50impl Scoreable for MpaReader<'_> {
51    fn score(mut src: ScopedStream<&mut MediaSourceStream<'_>>) -> Result<Score> {
52        // Read the sync word for the first (assumed) MPEG frame and try to parse it into a header.
53        let sync1 = header::read_frame_header_word_no_sync(&mut src)?;
54        let hdr1 = header::parse_frame_header(sync1)?;
55
56        // Since the first header was parsed successfully, this may be a MPEG audio format. However,
57        // if there is enough data left to read the frame body and another frame header, then a
58        // higher confidence may be gained. If there is not enough data left, return a partially
59        // confident score.
60        if src.bytes_available() < (hdr1.frame_size + header::MPEG_HEADER_LEN) as u64 {
61            return Ok(Score::Supported(127));
62        }
63
64        // Skip the frame body.
65        src.ignore_bytes(hdr1.frame_size as u64)?;
66
67        // Read another sync word for the second (assumed) MPEG frame.
68        let sync2 = header::read_frame_header_word_no_sync(&mut src)?;
69
70        // The second sync word should look like a sync word.
71        if !header::is_frame_header_word_synced(sync2) {
72            return Ok(Score::Unsupported);
73        }
74
75        // Try to parse the second sync word into a header.
76        let _ = header::parse_frame_header(sync2)?;
77
78        Ok(Score::Supported(255))
79    }
80}

160    fn next_packet(&mut self) -> Result<Option<Packet>> {
161        let (header, data) = loop {
162            // Read the next MPEG frame.
163            let (header, data) = match read_mpeg_frame(&mut self.reader) {
164                Ok(frame) => frame,
165                Err(Error::IoError(err)) if err.kind() == std::io::ErrorKind::UnexpectedEof => {
166                    // MPEG streams have no well-defined end, so when no more frames can be read,
167                    // consider the stream ended.
168                    return Ok(None);
169                }
170                Err(err) => return Err(err),
171            };
172
173            // Check if the packet contains a Xing, Info, or VBRI tag.
174            if is_maybe_info_tag(&data, &header) {
175                if try_read_info_tag(&data, &header).is_some() {
176                    // Discard the packet and tag since it was not at the start of the stream.
177                    warn!("found an unexpected xing tag, discarding");
178                    continue;
179                }
180            }
181            else if is_maybe_vbri_tag(&data, &header)
182                && try_read_vbri_tag(&data, &header).is_some()
183            {
184                // Discard the packet and tag since it was not at the start of the stream.
185                warn!("found an unexpected vbri tag, discarding");
186                continue;
187            }
188
189            break (header, data);
190        };
191
192        // The timestamp and duration for this packet.
193        let pts = self.next_packet_ts;
194        let dur = header.duration();
195
196        // Advance the next packet timestamp based on this packet's duration. If it saturates, then
197        // it is not possible to read further.
198        self.next_packet_ts = match self.next_packet_ts.checked_add(dur) {
199            Some(ts) => ts,
200            None => return Ok(None),
201        };
202
203        // Build the packet.
204        let packet = PacketBuilder::new()
205            .track_id(0)
206            .pts(pts)
207            .trimmed_dur(
208                dur,
209                self.tracks[0]
210                    .num_frames
211                    .map(Duration::from)
212                    .and_then(|dur| dur.timestamp_from(Timestamp::ZERO)),
213            )
214            .data(data)
215            .build();
216
217        Ok(Some(packet))
218    }

232    fn seek(&mut self, mode: SeekMode, to: SeekTo) -> Result<SeekedTo> {
233        const MAX_REF_FRAMES: usize = 4;
234        const REF_FRAMES_MASK: usize = MAX_REF_FRAMES - 1;
235
236        // Get the timestamp of the desired audio frame.
237        let required_ts = match to {
238            // Frame timestamp given.
239            SeekTo::Timestamp { ts, .. } => ts,
240            // Time value given, calculate frame timestamp using the timebase.
241            SeekTo::Time { time, .. } => {
242                // The timebase is required to calculate the timestamp.
243                let tb =
244                    self.tracks[0].time_base.ok_or(Error::SeekError(SeekErrorKind::Unseekable))?;
245
246                // If the timestamp overflows, the seek if out-of-range.
247                tb.calc_timestamp(time).ok_or(Error::SeekError(SeekErrorKind::OutOfRange))?
248            }
249        };
250
251        // Get the total duration of the track (excludes delay and padding).
252        let dur_ts = self.tracks[0].num_frames.map(Duration::from);
253
254        // Get the delay and padding values (or 0 if unknown).
255        let delay = self.tracks[0].delay.unwrap_or(0);
256        let padding = self.tracks[0].padding.unwrap_or(0);
257
258        // Calculate the valid timestamp bounds of the track.
259        let min_ts = Timestamp::from(-i64::from(delay));
260        let max_ts = dur_ts
261            .and_then(|dur| min_ts.checked_add(dur))
262            .and_then(|dur| dur.checked_add(Duration::from(delay + padding)));
263
264        // Ensure the seek position is within the bounds of the track.
265        if required_ts < min_ts {
266            return seek_error(SeekErrorKind::OutOfRange);
267        }
268        else if let Some(max_ts) = max_ts {
269            if required_ts > max_ts {
270                return seek_error(SeekErrorKind::OutOfRange);
271            }
272        }
273
274        let is_seekable = self.reader.is_seekable();
275
276        // If the stream is unseekable and the required timestamp in the past, then return an
277        // error, it is not possible to seek to it.
278        if !is_seekable && required_ts < self.next_packet_ts {
279            return seek_error(SeekErrorKind::ForwardOnly);
280        }
281
282        debug!("seeking to ts={required_ts}");
283
284        // Step 1
285        //
286        // In coarse seek mode, the underlying media source stream will be roughly seeked based on
287        // the required timestamp and the total duration of the media. Coarse seek mode requires a
288        // seekable stream because the total length in bytes of the stream is required.
289        //
290        // In accurate seek mode, the underlying media source stream will not be seeked unless the
291        // required timestamp is in the past, in which case the stream is seeked back to the start.
292        match mode {
293            SeekMode::Coarse if is_seekable => self.preseek_coarse(required_ts, min_ts, max_ts)?,
294            SeekMode::Accurate => self.preseek_accurate(required_ts, min_ts)?,
295            _ => (),
296        };
297
298        // Step 2
299        //
300        // Following the pre-seek operation above, parse MPEG frames (packets) one-by-one from the
301        // current position in the stream until the frame containing the desired timestamp is
302        // reached. For coarse seeks, this should only parse a few packets. For accurate seeks, the
303        // entire stream could potentially be parsed.
304        let mut frames: [FramePos; MAX_REF_FRAMES] = Default::default();
305        let mut n_parsed = 0;
306
307        loop {
308            // Sync to the next frame.
309            let sync = match header::sync_frame(&mut self.reader) {
310                Ok(sync) => sync,
311                Err(Error::IoError(err)) if err.kind() == std::io::ErrorKind::UnexpectedEof => {
312                    // MPEG streams have no well-defined end, so if no more frames can be read then
313                    // assume the seek position is out-of-range. This would normally only happen if
314                    // the duration of the track is unknown, or it is was longer than the track
315                    // actually is.
316                    return seek_error(SeekErrorKind::OutOfRange);
317                }
318                Err(err) => return Err(err),
319            };
320
321            // Parse the synced frame header.
322            let header = header::parse_frame_header(sync)?;
323
324            // Position of the frame header.
325            let pos = self.reader.pos() - header::MPEG_HEADER_LEN as u64;
326
327            // Calculate the duration of the frame.
328            let frame_dur = header.duration();
329
330            // Add the frame to the frame ring.
331            frames[n_parsed & REF_FRAMES_MASK] = FramePos { pos, ts: self.next_packet_ts };
332            n_parsed += 1;
333
334            let next_packet_ts = match self.next_packet_ts.checked_add(frame_dur) {
335                Some(ts) if ts <= required_ts => ts,
336                // The timestamp of the next MPEG frame exceeds the desired timestamp, or it
337                // exceeds the representable range. Rewind back to the start of the current frame
338                // and end the search.
339                _ => {
340                    // The main_data_begin offset is a negative offset from the frame's header to
341                    // where its main data begins. Therefore, for a decoder to properly decode this
342                    // frame, the reader must provide previous (reference) frames up-to and
343                    // including the frame that contains the first byte this frame's main_data.
344                    let main_data_begin = read_main_data_begin(&mut self.reader, &header)? as u64;
345
346                    debug!(
347                        "found frame with ts={} @ pos={} with main_data_begin={}",
348                        self.next_packet_ts, pos, main_data_begin
349                    );
350
351                    // The number of reference frames is 0 if main_data_begin is also 0. Otherwise,
352                    // attempt to find the first (oldest) reference frame, then select 1 frame
353                    // before that one to actually seek to.
354                    let mut n_ref_frames = 0;
355                    let mut ref_frame = &frames[(n_parsed - 1) & REF_FRAMES_MASK];
356
357                    if main_data_begin > 0 {
358                        // The maximum number of reference frames is limited to the number of frames
359                        // read and the number of previous frames recorded.
360                        let max_ref_frames = std::cmp::min(n_parsed, frames.len());
361
362                        while n_ref_frames < max_ref_frames {
363                            ref_frame = &frames[(n_parsed - n_ref_frames - 1) & REF_FRAMES_MASK];
364
365                            if pos - ref_frame.pos >= main_data_begin {
366                                break;
367                            }
368
369                            n_ref_frames += 1;
370                        }
371
372                        debug!(
373                            "will seek -{} frame(s) to ts={} @ pos={} (-{} bytes)",
374                            n_ref_frames,
375                            ref_frame.ts,
376                            ref_frame.pos,
377                            pos - ref_frame.pos
378                        );
379                    }
380
381                    // Do the actual seek to the reference frame.
382                    self.reader.seek_buffered(ref_frame.pos);
383
384                    self.next_packet_ts = ref_frame.ts;
385                    break;
386                }
387            };
388
389            // Otherwise, ignore the frame body.
390            self.reader.ignore_bytes(header.frame_size as u64)?;
391
392            // Increment the timestamp for the next packet.
393            self.next_packet_ts = next_packet_ts;
394        }
395
396        debug!(
397            "seeked to ts={} (delta={})",
398            self.next_packet_ts,
399            self.next_packet_ts.saturating_delta(required_ts),
400        );
401
402        Ok(SeekedTo { track_id: 0, required_ts, actual_ts: self.next_packet_ts })
403    }

413impl<'s> MpaReader<'s> {
414    pub fn try_new(mut mss: MediaSourceStream<'s>, opts: FormatOptions) -> Result<Self> {
415        // Try to read the first MPEG frame.
416        let (header, packet) = read_mpeg_frame_strict(&mut mss)?;
417
418        // Use the header to populate the codec parameters.
419        let mut codec_params = AudioCodecParameters::new();
420
421        codec_params
422            .for_codec(header.codec())
423            .with_sample_rate(header.sample_rate)
424            .with_channels(header.channel_mode.channels());
425
426        // Create the track.
427        let mut track = Track::new(0);
428
429        track.with_codec_params(CodecParameters::Audio(codec_params));
430
431        // Check if there is a Xing/Info tag contained in the first frame.
432        if let Some(info_tag) = try_read_info_tag(&packet, &header) {
433            // The LAME tag contains ReplayGain and padding information.
434            if let Some(lame_tag) = info_tag.lame {
435                track.with_delay(lame_tag.enc_delay).with_padding(lame_tag.enc_padding);
436            }
437
438            // The base Xing/Info tag may contain the number of MPEG frames.
439            if let Some(num_mpeg_frames) = info_tag.num_frames {
440                info!("using xing header for duration");
441
442                // The total number of audio frames including delay and padding frames.
443                let num_frames = u64::from(num_mpeg_frames) * u64::from(header.num_frames());
444
445                // Remove delay and padding frames.
446                let discard = track.delay.unwrap_or(0) + track.padding.unwrap_or(0);
447
448                track.with_num_frames(num_frames.saturating_sub(u64::from(discard)));
449            }
450        }
451        else if let Some(vbri_tag) = try_read_vbri_tag(&packet, &header) {
452            info!("using vbri header for duration");
453
454            let num_frames = u64::from(vbri_tag.num_mpeg_frames) * u64::from(header.num_frames());
455
456            // Check if there is a VBRI tag.
457            track.with_num_frames(num_frames);
458        }
459        else {
460            // The first frame was not a Xing/Info header, rewind back to the start of the frame so
461            // that it may be decoded.
462            mss.seek_buffered_rev(MPEG_HEADER_LEN + header.frame_size);
463
464            // Likely not a VBR file, so estimate the duration if seekable.
465            if mss.is_seekable() {
466                info!("estimating duration from bitrate, may be inaccurate for vbr files");
467
468                if let Some(n_mpeg_frames) = estimate_num_mpeg_frames(&mut mss) {
469                    track.with_num_frames(n_mpeg_frames * u64::from(header.num_frames()));
470                }
471            }
472        }
473
474        if let Some(num_frames) = track.num_frames {
475            // Duration equals the number of frames because the timebase is always 1 / sample rate.
476            track.with_duration(Duration::from(num_frames));
477        }
478
479        let first_packet_pos = mss.pos();
480        let next_packet_ts = Timestamp::from(-i64::from(track.delay.unwrap_or(0)));
481
482        Ok(MpaReader {
483            reader: mss,
484            media_info: MediaInfo::from_track(&track),
485            tracks: vec![track],
486            chapters: opts.external_data.chapters,
487            metadata: opts.external_data.metadata.unwrap_or_default(),
488            first_packet_pos,
489            next_packet_ts,
490        })
491    }

493    /// Seeks the media source stream to a byte position roughly where the packet with the required
494    /// timestamp should be located.
495    fn preseek_coarse(
496        &mut self,
497        required_ts: Timestamp,
498        min_ts: Timestamp,
499        max_ts: Option<Timestamp>,
500    ) -> Result<()> {
501        // Get the length in bytes of the stream's audio data. It is not possible to coarsely seek
502        // without knowing this.
503        let audio_byte_len = match self.reader.byte_len() {
504            Some(byte_len) => u128::from(byte_len - self.first_packet_pos),
505            None => return seek_error(SeekErrorKind::Unseekable),
506        };
507
508        // It is not possible to coarse seek without knowing both the upper and lower timestamp
509        // bounds, and the lower timestamp bound must be <= the upper timestamp bound.
510        let max_ts = match max_ts {
511            Some(max_ts) if max_ts >= min_ts => max_ts,
512            _ => return seek_error(SeekErrorKind::Unseekable),
513        };
514
515        // Call invariants.
516        debug_assert!(min_ts <= required_ts);
517        debug_assert!(required_ts <= max_ts);
518
519        // The total duration of the track.
520        let total_dur = u128::from(max_ts.abs_delta(min_ts).get());
521
522        // Calculate, roughly, the byte position of where the MPEG frame containing the required
523        // timestamp should be in the media source stream relative to the start of the audio data.
524        let seek_pos_rel =
525            (u128::from(required_ts.abs_delta(min_ts).get()) * audio_byte_len) / total_dur;
526
527        // It is preferable to return a packet with a timestamp before the requested timestamp.
528        // Therefore, subtract the maximum packet size from the position found above to ensure this.
529        let seek_pos = (seek_pos_rel + u128::from(self.first_packet_pos))
530            .saturating_sub(MAX_MPEG_FRAME_SIZE as u128);
531
532        // Seek the media source stream.
533        self.reader.seek(SeekFrom::Start(
534            seek_pos.try_into().map_err(|_| Error::SeekError(SeekErrorKind::OutOfRange))?,
535        ))?;
536
537        // Resync to the start of the next MPEG frame.
538        let (header, _) = read_mpeg_frame_strict(&mut self.reader)?;
539
540        // The byte position of the next MPEG frame relative to the start of the audio data.
541        let audio_byte_pos = u128::from(self.reader.pos() - self.first_packet_pos);
542
543        // Calculate, roughly, the duration from the start of the audio of the packet based on the
544        // byte position after resync.
545        let dur_to_pkt = Duration::from(((audio_byte_pos * total_dur) / audio_byte_len) as u64);
546
547        // Assume the duration of a packet remains constant throughout the stream (not a
548        // guarantee, but usually the case).
549        let pkt_dur = header.duration();
550
551        // Compute the timestamp of the next packet. First align the estimated duration from the
552        // start of the audio to a packet boundary. Then add to the lower-bound timestamp. In
553        // theory, this should never exceed the upper-bound timestamp.
554        //
555        self.next_packet_ts = min_ts
556            .checked_add(
557                dur_to_pkt.align_down(pkt_dur).expect("packet duration is always non-zero"),
558            )
559            .unwrap_or(max_ts);
560
561        Ok(())
562    }
563
564    /// Seeks the media source stream back to the start of the first packet if the required
565    /// timestamp is in the past.
566    fn preseek_accurate(&mut self, required_ts: Timestamp, min_ts: Timestamp) -> Result<()> {
567        if required_ts < self.next_packet_ts {
568            let seeked_pos = self.reader.seek(SeekFrom::Start(self.first_packet_pos))?;
569
570            // Since the elementary stream has no timestamp information, the position seeked
571            // to must be exactly as requested.
572            if seeked_pos != self.first_packet_pos {
573                return seek_error(SeekErrorKind::Unseekable);
574            }
575
576            // Successfuly seeked to the start of the stream, reset the next packet timestamp.
577            self.next_packet_ts = min_ts;
578        }
579
580        Ok(())
581    }
582}

584/// Reads a MPEG frame and returns the header and buffer.
585fn read_mpeg_frame(reader: &mut MediaSourceStream<'_>) -> Result<(FrameHeader, Vec<u8>)> {
586    let (header, header_word) = loop {
587        // Sync to the next frame header.
588        let sync = header::sync_frame(reader)?;
589
590        // Parse the frame header fully.
591        if let Ok(header) = header::parse_frame_header(sync) {
592            break (header, sync);
593        }
594
595        warn!("invalid mpeg audio header");
596    };
597
598    // Allocate frame buffer.
599    let mut packet = vec![0u8; MPEG_HEADER_LEN + header.frame_size];
600    packet[0..MPEG_HEADER_LEN].copy_from_slice(&header_word.to_be_bytes());
601
602    // Read the frame body.
603    reader.read_buf_exact(&mut packet[MPEG_HEADER_LEN..])?;
604
605    // Return the parsed header and packet body.
606    Ok((header, packet))
607}

609/// Reads a MPEG frame and checks if the next frame begins after the packet.
610fn read_mpeg_frame_strict(reader: &mut MediaSourceStream<'_>) -> Result<(FrameHeader, Vec<u8>)> {
611    loop {
612        // Read the next MPEG frame.
613        let (header, packet) = read_mpeg_frame(reader)?;
614
615        // Get the position before trying to read the next header.
616        let pos = reader.pos();
617
618        // Read a sync word from the stream. If this read fails then the file may have ended and
619        // this check cannot be performed.
620        if let Ok(sync) = header::read_frame_header_word_no_sync(reader) {
621            // If the stream is not synced to the next frame's sync word, or the next frame header
622            // is not parseable or similar to the current frame header, then reject the current
623            // packet since the stream likely synced to random data.
624            if !header::is_frame_header_word_synced(sync) || !is_frame_header_similar(&header, sync)
625            {
626                warn!("skipping junk at {} bytes", pos - packet.len() as u64);
627
628                // Seek back to the second byte of the rejected packet to prevent syncing to the
629                // same spot again.
630                reader.seek_buffered_rev(packet.len() + MPEG_HEADER_LEN - 1);
631                continue;
632            }
633        }
634
635        // Jump back to the position before the next header was read.
636        reader.seek_buffered(pos);
637
638        break Ok((header, packet));
639    }
640}
641
642/// Check if a sync word parses to a frame header that is similar to the one provided.
643fn is_frame_header_similar(header: &FrameHeader, sync: u32) -> bool {
644    if let Ok(candidate) = header::parse_frame_header(sync) {
645        if header.version == candidate.version
646            && header.layer == candidate.layer
647            && header.sample_rate == candidate.sample_rate
648            && header.n_channels() == candidate.n_channels()
649        {
650            return true;
651        }
652    }
653
654    false
655}

663/// Reads the main_data_begin field from the side information of a MPEG audio frame.
664fn read_main_data_begin<B: ReadBytes>(reader: &mut B, header: &FrameHeader) -> Result<u16> {
665    // After the head the optional CRC is present.
666    if header.has_crc {
667        let _crc = reader.read_be_u16()?;
668    }
669
670    // For MPEG version 1 the first 9 bits is main_data_begin.
671    let main_data_begin = if header.is_mpeg1() {
672        reader.read_be_u16()? >> 7
673    }
674    // For MPEG version 2 the first 8 bits is main_data_begin.
675    else {
676        u16::from(reader.read_u8()?)
677    };
678
679    Ok(main_data_begin)
680}

682/// Estimates the total number of MPEG frames in the media source stream.
683fn estimate_num_mpeg_frames(reader: &mut MediaSourceStream<'_>) -> Option<u64> {
684    const MAX_FRAMES: u32 = 16;
685    const MAX_LEN: usize = 16 * 1024;
686
687    // Macro to convert a Result to Option, and break a loop on exit.
688    macro_rules! break_on_err {
689        ($expr:expr) => {
690            match $expr {
691                Ok(a) => a,
692                _ => break None,
693            }
694        };
695    }
696
697    let start_pos = reader.pos();
698
699    let mut total_frame_len = 0;
700    let mut total_frames = 0;
701
702    let total_len = match reader.byte_len() {
703        Some(len) => len - start_pos,
704        _ => return None,
705    };
706
707    let num_mpeg_frames = loop {
708        // Read the frame header.
709        let header_val = break_on_err!(reader.read_be_u32());
710
711        // Parse the frame header.
712        let header = break_on_err!(header::parse_frame_header(header_val));
713
714        // Tabulate the size.
715        total_frame_len += MPEG_HEADER_LEN + header.frame_size;
716        total_frames += 1;
717
718        // Ignore the frame body.
719        break_on_err!(reader.ignore_bytes(header.frame_size as u64));
720
721        // Read up-to 16 frames, or 16kB, then calculate the average MPEG frame length, and from
722        // that, the total number of MPEG frames.
723        if total_frames > MAX_FRAMES || total_frame_len > MAX_LEN {
724            let avg_mpeg_frame_len = total_frame_len as f64 / total_frames as f64;
725            break Some((total_len as f64 / avg_mpeg_frame_len) as u64);
726        }
727    };
728
729    // Rewind back to the first frame seen upon entering this function.
730    reader.seek_buffered_rev((reader.pos() - start_pos) as usize);
731
732    num_mpeg_frames
733}

739#[allow(dead_code)]
740struct LameTag {
741    encoder: String,
742    replaygain_peak: Option<f32>,

744    replaygain_audiophile: Option<f32>,

746    enc_padding: u32,

749/// The Xing/Info time additional information for regarding a MP3 file.
750#[allow(dead_code)]
751struct XingInfoTag {
752    num_frames: Option<u32>,
753    num_bytes: Option<u32>,
754    toc: Option<[u8; 100]>,
755    quality: Option<u32>,
756    is_cbr: bool,
757    lame: Option<LameTag>,
758}

760/// Try to read a Xing/Info tag from the provided MPEG frame.
761fn try_read_info_tag(buf: &[u8], header: &FrameHeader) -> Option<XingInfoTag> {
762    // The Info header is a completely optional piece of information. Therefore, flatten an error
763    // reading the tag into a None.
764    try_read_info_tag_inner(buf, header).ok().flatten()
765}
766
767fn try_read_info_tag_inner(buf: &[u8], header: &FrameHeader) -> Result<Option<XingInfoTag>> {
768    // Do a quick check that this is a Xing/Info tag.
769    if !is_maybe_info_tag(buf, header) {
770        return Ok(None);
771    }
772
773    // The position of the Xing/Info tag relative to the end of the header. This is equal to the
774    // side information length for the frame. The CRC is not included in this offset calculation.
775    let offset = MPEG_HEADER_LEN + header.side_info_len();
776
777    // Start the CRC with the header and side information.
778    let mut crc16 = Crc16AnsiLe::new(0);
779    crc16.process_buf_bytes(&buf[..offset]);
780
781    // Start reading the Xing/Info tag after the side information.
782    let mut reader = MonitorStream::new(BufReader::new(&buf[offset..]), crc16);
783
784    // Check for Xing/Info header.
785    let id = reader.read_quad_bytes()?;
786
787    if id != XING_TAG_ID && id != INFO_TAG_ID {
788        return Ok(None);
789    }
790
791    // The "Info" id is used for CBR files.
792    let is_cbr = id == INFO_TAG_ID;
793
794    // Flags indicates what information is provided in this Xing/Info tag.
795    let flags = reader.read_be_u32()?;
796
797    let num_frames = if flags & 0x1 != 0 { Some(reader.read_be_u32()?) } else { None };
798
799    let num_bytes = if flags & 0x2 != 0 { Some(reader.read_be_u32()?) } else { None };
800
801    let toc = if flags & 0x4 != 0 {
802        let mut toc = [0; 100];
803        reader.read_buf_exact(&mut toc)?;
804        Some(toc)
805    }
806    else {
807        None
808    };
809
810    let quality = if flags & 0x8 != 0 { Some(reader.read_be_u32()?) } else { None };
811
812    /// The full LAME extension size.
813    const LAME_EXT_LEN: u64 = 36;
814    /// The minimal LAME extension size up-to the encode delay & padding fields.
815    const MIN_LAME_EXT_LEN: u64 = 24;
816
817    // The LAME extension may not always be present, or complete. The important fields in the
818    // extension are within the first 24 bytes. Therefore, try to read those if they're available.
819    let lame = if reader.inner().bytes_available() >= MIN_LAME_EXT_LEN {
820        // Encoder string.
821        let mut encoder = [0; 9];
822        reader.read_buf_exact(&mut encoder)?;
823
824        // Revision.
825        let _revision = reader.read_u8()?;
826
827        // Lowpass filter value.
828        let _lowpass = reader.read_u8()?;
829
830        // Replay gain peak in 9.23 (bit) fixed-point format.
831        let replaygain_peak = match reader.read_be_u32()? {
832            0 => None,
833            peak => Some(32767.0 * (peak as f32 / 2.0f32.powi(23))),
834        };
835
836        // Radio replay gain.
837        let replaygain_radio = parse_lame_tag_replaygain(reader.read_be_u16()?, 1);
838
839        // Audiophile replay gain.
840        let replaygain_audiophile = parse_lame_tag_replaygain(reader.read_be_u16()?, 2);
841
842        // Encoding flags & ATH type.
843        let _encoding_flags = reader.read_u8()?;
844
845        // Arbitrary bitrate.
846        let _abr = reader.read_u8()?;
847
848        let (enc_delay, enc_padding) = {
849            let trim = reader.read_be_u24()?;
850
851            if encoder[..4] == *b"LAME" || encoder[..4] == *b"Lavf" || encoder[..4] == *b"Lavc" {
852                let delay = 528 + 1 + (trim >> 12);
853                let padding = trim & ((1 << 12) - 1);
854
855                (delay, padding.saturating_sub(528 + 1))
856            }
857            else {
858                (0, 0)
859            }
860        };
861
862        // If possible, attempt to read the extra fields of the extension if they weren't
863        // truncated.
864        let crc = if reader.inner().bytes_available() >= LAME_EXT_LEN - MIN_LAME_EXT_LEN {
865            // Flags.
866            let _misc = reader.read_u8()?;
867
868            // MP3 gain.
869            let _mp3_gain = reader.read_u8()?;
870
871            // Preset and surround info.
872            let _surround_info = reader.read_be_u16()?;
873
874            // Music length.
875            let _music_len = reader.read_be_u32()?;
876
877            // Music (audio) CRC.
878            let _music_crc = reader.read_be_u16()?;
879
880            // The tag CRC. LAME always includes this CRC regardless of the protection bit, but
881            // other encoders may only do so if the protection bit is set.
882            if header.has_crc || encoder[..4] == *b"LAME" {
883                // Read the CRC using the inner reader to not change the computed CRC.
884                Some(reader.inner_mut().read_be_u16()?)
885            }
886            else {
887                // No CRC is present.
888                None
889            }
890        }
891        else {
892            // The tag is truncated. No CRC will be present.
893            info!("xing tag lame extension is truncated");
894            None
895        };
896
897        // If there was no CRC written, then assume the tag is correct. Otherwise, use the CRC.
898        // Accept a written CRC of 0 which defacto means to ignore the CRC.
899        let is_tag_ok = crc.is_none_or(|crc| crc == 0 || crc == reader.monitor().crc());
900
901        if is_tag_ok {
902            // The CRC matched or is not present.
903            Some(LameTag {
904                encoder: String::from_utf8_lossy(&encoder).into(),
905                replaygain_peak,
906                replaygain_radio,
907                replaygain_audiophile,
908                enc_delay,
909                enc_padding,
910            })
911        }
912        else {
913            // The CRC did not match, this is probably not a LAME tag.
914            warn!("xing tag lame extension crc mismatch");
915            None
916        }
917    }
918    else {
919        // Frame not large enough for a LAME tag.
920        info!("xing tag too small for lame extension");
921        None
922    };
923
924    Ok(Some(XingInfoTag { num_frames, num_bytes, toc, quality, is_cbr, lame }))
925}

927fn parse_lame_tag_replaygain(value: u16, expected_name: u8) -> Option<f32> {
928    // The 3 most-significant bits are the name code.
929    let name = ((value & 0xe000) >> 13) as u8;
930
931    if name == expected_name {
932        let gain = (value & 0x01ff) as f32 / 10.0;
933        Some(if value & 0x200 != 0 { -gain } else { gain })
934    }
935    else {
936        None
937    }
938}

940/// Perform a fast check to see if the packet contains a Xing/Info tag. If this returns true, the
941/// packet should be parsed fully to ensure it is in fact a tag.
942fn is_maybe_info_tag(buf: &[u8], header: &FrameHeader) -> bool {
943    const MIN_XING_TAG_LEN: usize = 8;
944
945    // Only supported with layer 3 packets.
946    if header.layer != MpegLayer::Layer3 {
947        return false;
948    }
949
950    // The position of the Xing/Info tag relative to the start of the packet. This is equal to the
951    // side information length for the frame. The CRC is not included in this offset calculation.
952    let offset = MPEG_HEADER_LEN + header.side_info_len();
953
954    // The packet must be big enough to contain a tag.
955    if buf.len() < offset + MIN_XING_TAG_LEN {
956        return false;
957    }
958
959    // The tag ID must be present and correct.
960    let id = &buf[offset..offset + 4];
961
962    if id != XING_TAG_ID && id != INFO_TAG_ID {
963        return false;
964    }
965
966    // The side information, which follows the header and optional CRC, should be zeroed.
967    !buf[header.header_size()..offset].iter().any(|&b| b != 0)
968}

979/// Try to read a VBRI tag from the provided MPEG frame.
980fn try_read_vbri_tag(buf: &[u8], header: &FrameHeader) -> Option<VbriTag> {
981    // The VBRI header is a completely optional piece of information. Therefore, flatten an error
982    // reading the tag into a None.
983    try_read_vbri_tag_inner(buf, header).ok().flatten()
984}
985
986fn try_read_vbri_tag_inner(buf: &[u8], header: &FrameHeader) -> Result<Option<VbriTag>> {
987    // Do a quick check that this is a VBRI tag.
988    if !is_maybe_vbri_tag(buf, header) {
989        return Ok(None);
990    }
991
992    let mut reader = BufReader::new(buf);
993
994    // The VBRI tag is always 32 bytes after the header.
995    reader.ignore_bytes(MPEG_HEADER_LEN as u64 + 32)?;
996
997    // Check for the VBRI signature.
998    let id = reader.read_quad_bytes()?;
999
1000    if id != VBRI_TAG_ID {
1001        return Ok(None);
1002    }
1003
1004    // The version is always 1.
1005    let version = reader.read_be_u16()?;
1006
1007    if version != 1 {
1008        return Ok(None);
1009    }
1010
1011    // Delay is a 2-byte big-endiann floating point value?
1012    let _delay = reader.read_be_u16()?;
1013    let _quality = reader.read_be_u16()?;
1014
1015    let num_bytes = reader.read_be_u32()?;
1016    let num_mpeg_frames = reader.read_be_u32()?;
1017
1018    Ok(Some(VbriTag { num_bytes, num_mpeg_frames }))
1019}

1021/// Perform a fast check to see if the packet contains a VBRI tag. If this returns true, the
1022/// packet should be parsed fully to ensure it is in fact a tag.
1023fn is_maybe_vbri_tag(buf: &[u8], header: &FrameHeader) -> bool {
1024    const MIN_VBRI_TAG_LEN: usize = 26;
1025    const VBRI_TAG_OFFSET: usize = MPEG_HEADER_LEN + 32;
1026
1027    // Only supported with layer 3 packets.
1028    if header.layer != MpegLayer::Layer3 {
1029        return false;
1030    }
1031
1032    // The packet must be big enough to contain a tag.
1033    if buf.len() < VBRI_TAG_OFFSET + MIN_VBRI_TAG_LEN {
1034        return false;
1035    }
1036
1037    // The tag ID must be present and correct.
1038    let id = &buf[VBRI_TAG_OFFSET..VBRI_TAG_OFFSET + 4];
1039
1040    if id != VBRI_TAG_ID {
1041        return false;
1042    }
1043
1044    // The bytes preceeding the VBRI tag (mostly the side information) should be all 0.
1045    !buf[header.header_size()..VBRI_TAG_OFFSET].iter().any(|&b| b != 0)
1046}
