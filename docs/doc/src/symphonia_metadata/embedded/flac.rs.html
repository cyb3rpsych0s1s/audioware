35/// Converts a string of bytes to an ASCII string if all characters are within the printable ASCII
36/// range. If a null byte is encounted, the string terminates at that point.
37fn printable_ascii_to_string(bytes: &[u8]) -> Option<String> {
38    let mut result = String::with_capacity(bytes.len());
39
40    for c in bytes {
41        match c {
42            0x00 => break,
43            0x20..=0x7e => result.push(char::from(*c)),
44            _ => return None,
45        }
46    }
47
48    Some(result)
49}

61/// Read a picture metadata block.
62pub fn read_flac_picture_block<B: ReadBytes>(reader: &mut B) -> Result<Visual> {
63    let type_enc = reader.read_be_u32()?;
64
65    // Read the Media Type length in bytes.
66    // TODO: Apply a limit.
67    let media_type_len = reader.read_be_u32()? as usize;
68
69    // Read the Media Type bytes
70    let media_type_buf = reader.read_boxed_slice_exact(media_type_len)?;
71
72    // Convert Media Type bytes to an ASCII string. Non-printable ASCII characters are invalid.
73    let media_type = match printable_ascii_to_string(&media_type_buf) {
74        Some(s) => {
75            // Return None if the media-type string is empty.
76            Some(s).filter(|s| !s.is_empty())
77        }
78        None => return decode_error("meta (flac): picture mime-type contains invalid characters"),
79    };
80
81    let mut tags = vec![];
82
83    // Read the description length in bytes.
84    // TODO: Apply a limit.
85    let desc_len = reader.read_be_u32()? as usize;
86
87    // Read the description bytes.
88    let desc_buf = reader.read_boxed_slice_exact(desc_len)?;
89
90    // Convert to a UTF-8 string.
91    let desc = String::from_utf8_lossy(&desc_buf);
92
93    if !desc.is_empty() {
94        let desc = Arc::new(desc.into_owned());
95        let tag =
96            Tag::new_from_parts("DESCRIPTION", desc.clone(), Some(StandardTag::Description(desc)));
97
98        tags.push(tag);
99    }
100
101    // Read the width, and height of the visual.
102    let width = reader.read_be_u32()?;
103    let height = reader.read_be_u32()?;
104
105    // If either the width or height is 0, then the size is invalid.
106    let dimensions = if width > 0 && height > 0 { Some(Size { width, height }) } else { None };
107
108    // Read bits-per-pixel of the visual.
109    let _bits_per_pixel = NonZeroU8::new(reader.read_be_u32()? as u8);
110
111    // Indexed colours is only valid for image formats that use an indexed colour palette. If it is
112    // 0, the image does not used indexed colours.
113    let _color_mode = reader.read_be_u32()?;
114
115    // Read the image data length in bytes.
116    // TODO: Apply a limit.
117    let data_len = reader.read_be_u32()? as usize;
118
119    // Read the image data.
120    let data = reader.read_boxed_slice_exact(data_len)?;
121
122    // Try to detect the image characteristics from the image data. Detect image characteristics
123    // will be preferred over what's been stated in the picture block.
124    let image_info = try_get_image_info(&data);
125
126    Ok(Visual {
127        media_type: image_info.as_ref().map(|info| info.media_type.clone()).or(media_type),
128        dimensions: image_info.as_ref().map(|info| info.dimensions).or(dimensions),
129        color_mode: image_info.as_ref().map(|info| info.color_mode),
130        usage: get_visual_key_from_picture_type(type_enc),
131        tags,
132        data,
133    })
134}

136/// Read a seek table metadata block as a seek index.
137pub fn read_flac_seektable_block<B: ReadBytes>(
138    reader: &mut B,
139    block_length: u32,
140) -> Result<SeekIndex> {
141    // The number of seek table entries is always the metadata block length divided by the length of
142    // a single entry, 18 bytes.
143    let count = block_length / 18;
144
145    let mut index = SeekIndex::new();
146
147    // Read each entry in the seektable.
148    for _ in 0..count {
149        // The sample number (timestamp) of the first sample in the target frame.
150        let sample = reader.read_be_u64()?;
151
152        // A sample number of 0xFFFFFFFFFFFFFFFF is designated as a placeholder and indicates the
153        // entry should be ignored by decoders. However, since timestamps > 0x7FFFFFFFFFFFFFFF are
154        // unrepresentable in Symphonia, all of these entries will also be ignored.
155        if let Ok(sample) = sample.try_into() {
156            // The byte offset of the target frame.
157            let offset = reader.read_be_u64()?;
158            // The number of samples in the target frame.
159            let num_samples = reader.read_be_u16()?;
160            index.insert(sample, offset, u32::from(num_samples));
161        }
162        else {
163            reader.ignore_bytes(10)?
164        }
165    }
166
167    Ok(index)
168}

170/// Read a vendor-specific application metadata block as vendor data.
171pub fn read_flac_application_block<B: ReadBytes>(
172    reader: &mut B,
173    block_length: u32,
174) -> Result<VendorDataAttachment> {
175    // Read the application identifier. Usually this is just 4 ASCII characters, but it is not
176    // limited to that. Non-printable ASCII characters must be escaped to create a valid UTF8
177    // string.
178    let ident = escape_identifier(&reader.read_quad_bytes()?);
179    let data = reader.read_boxed_slice_exact(block_length as usize - 4)?;
180    Ok(VendorDataAttachment { ident, data })
181}

183/// Read a cuesheet metadata block as a chapter group.
184pub fn read_flac_cuesheet_block<B: ReadBytes>(
185    reader: &mut B,
186    tb: TimeBase,
187) -> Result<ChapterGroup> {
188    // Read cuesheet catalog number. The catalog number only allows printable ASCII characters.
189    let mut catalog_number_buf = vec![0u8; 128];
190    reader.read_buf_exact(&mut catalog_number_buf)?;
191
192    // Read the catalog number, and store it in a Tag to be attached to the chapter group.
193    let catalog_number = match printable_ascii_to_string(&catalog_number_buf) {
194        Some(num) => {
195            let num = Arc::new(num);
196            Tag::new_from_parts("CATALOG", num.clone(), Some(StandardTag::IdentCatalogNumber(num)))
197        }
198        None => return decode_error("flac: cuesheet catalog number contains invalid characters"),
199    };
200
201    // Number of lead-in samples.
202    let n_lead_in_samples = reader.read_be_u64()?;
203
204    // Next bit is set for CD-DA cuesheets.
205    let is_cdda = (reader.read_u8()? & 0x80) == 0x80;
206
207    // Lead-in should be non-zero only for CD-DA cuesheets.
208    if !is_cdda && n_lead_in_samples > 0 {
209        return decode_error("flac: cuesheet lead-in samples should be zero if not CD-DA");
210    }
211
212    // Next 258 bytes (read as 129 u16's) must be zero.
213    for _ in 0..129 {
214        if reader.read_be_u16()? != 0 {
215            return decode_error("flac: cuesheet reserved bits should be zero");
216        }
217    }
218
219    let n_tracks = reader.read_u8()?;
220
221    // There should be at-least one track in the cuesheet.
222    if n_tracks == 0 {
223        return decode_error("flac: cuesheet must have at-least one track");
224    }
225
226    // CD-DA cuesheets must have no more than 100 tracks (99 audio tracks + lead-out track)
227    if is_cdda && n_tracks > 100 {
228        return decode_error("flac: cuesheets for CD-DA must not have more than 100 tracks");
229    }
230
231    let mut group = ChapterGroup {
232        items: Vec::with_capacity(usize::from(n_tracks)),
233        tags: vec![catalog_number],
234        visuals: Vec::new(),
235    };
236
237    for _ in 0..n_tracks {
238        group.items.push(read_flac_cuesheet_track(reader, is_cdda, tb)?);
239    }
240
241    Ok(group)
242}

244fn read_flac_cuesheet_track<B: ReadBytes>(
245    reader: &mut B,
246    is_cdda: bool,
247    tb: TimeBase,
248) -> Result<ChapterGroupItem> {
249    let n_offset_samples = reader.read_be_u64()?;
250
251    // For a CD-DA cuesheet, the track sample offset is the same as the first index (INDEX 00 or
252    // INDEX 01) on the CD. Therefore, the offset must be a multiple of 588 samples
253    // (588 samples = 44100 samples/sec * 1/75th of a sec).
254    if is_cdda && n_offset_samples % 588 != 0 {
255        return decode_error(
256            "flac: cuesheet track sample offset is not a multiple of 588 for CD-DA",
257        );
258    }
259
260    let number = reader.read_u8()?;
261
262    // A track number of 0 is disallowed in all cases. For CD-DA cuesheets, track 0 is reserved for
263    // lead-in.
264    if number == 0 {
265        return decode_error("flac: cuesheet track number of 0 not allowed");
266    }
267
268    // For CD-DA cuesheets, only track numbers 1-99 are allowed for regular tracks and 170 for
269    // lead-out.
270    if is_cdda && number > 99 && number != 170 {
271        return decode_error(
272            "flac: cuesheet track numbers greater than 99 are not allowed for CD-DA",
273        );
274    }
275
276    let mut isrc_buf = vec![0u8; 12];
277    reader.read_buf_exact(&mut isrc_buf)?;
278
279    let isrc = match printable_ascii_to_string(&isrc_buf) {
280        Some(num) => {
281            let num = Arc::new(num);
282            Tag::new_from_parts("ISRC", num.clone(), Some(StandardTag::IdentIsrc(num)))
283        }
284        None => return decode_error("flac: cuesheet track ISRC contains invalid characters"),
285    };
286
287    // Next 14 bytes are reserved. However, the first two bits are flags. Consume the reserved bytes
288    // in u16 chunks a minor performance improvement.
289    let flags = reader.read_be_u16()?;
290
291    // These values are contained in the Cuesheet but have no analogue in Symphonia.
292    let _is_audio = (flags & 0x8000) == 0x0000;
293    let _use_pre_emphasis = (flags & 0x4000) == 0x4000;
294
295    if flags & 0x3fff != 0 {
296        return decode_error("flac: cuesheet track reserved bits should be zero");
297    }
298
299    // Consume the remaining 12 bytes read in 3 u32 chunks.
300    for _ in 0..3 {
301        if reader.read_be_u32()? != 0 {
302            return decode_error("flac: cuesheet track reserved bits should be zero");
303        }
304    }
305
306    let n_indicies = reader.read_u8()? as usize;
307
308    // For CD-DA cuesheets, the track index cannot exceed 100 indicies.
309    if is_cdda && n_indicies > 100 {
310        return decode_error("flac: cuesheet track indicies cannot exceed 100 for CD-DA");
311    }
312
313    // If the track contains indicies, then one chapter will be created per index, and a chapter
314    // group returned.
315    if n_indicies > 0 {
316        let mut group = ChapterGroup {
317            items: Vec::with_capacity(n_indicies),
318            tags: vec![isrc],
319            visuals: vec![],
320        };
321
322        // Add each index as a chapter.
323        for _ in 0..n_indicies {
324            let index = read_flac_cuesheet_track_index(reader, is_cdda, n_offset_samples, tb)?;
325
326            group.items.push(ChapterGroupItem::Chapter(index));
327        }
328
329        Ok(ChapterGroupItem::Group(group))
330    }
331    else {
332        let start_ts = n_offset_samples.try_into().unwrap_or_else(|_| {
333            warn!("cuesheet track index offset too large, clamping to maximum");
334            i64::MAX
335        });
336
337        // If the track has no indicies, return a single chapter.
338        let chapter = Chapter {
339            start_time: tb.calc_time_saturating(Timestamp::from(start_ts)),
340            end_time: None,
341            start_byte: None,
342            end_byte: None,
343            tags: vec![isrc],
344            visuals: vec![],
345        };
346
347        Ok(ChapterGroupItem::Chapter(chapter))
348    }
349}

351fn read_flac_cuesheet_track_index<B: ReadBytes>(
352    reader: &mut B,
353    is_cdda: bool,
354    track_ts: u64,
355    tb: TimeBase,
356) -> Result<Chapter> {
357    let n_offset_samples = reader.read_be_u64()?;
358    let idx_number_raw = reader.read_be_u32()?;
359
360    // CD-DA track index points must have a sample offset that is a multiple of 588 samples
361    // (588 samples = 44100 samples/sec * 1/75th of a sec).
362    if is_cdda && n_offset_samples % 588 != 0 {
363        return decode_error(
364            "flac: cuesheet track index point sample offset is not a multiple of 588 for CD-DA",
365        );
366    }
367
368    // Lower 24 bits are reserved and should be 0.
369    if idx_number_raw & 0x00ff_ffff != 0 {
370        return decode_error("flac: cuesheet track index reserved bits should be 0");
371    }
372
373    // Upper 8 bits is the index number.
374    // TODO: Should be 0 or 1 for the first index for CD-DA. It should also not exceed 100 for
375    //       CD-DA.
376    let idx_number = ((idx_number_raw & 0xff00_0000) >> 24) as u8;
377
378    // Calculate the track index's starting timestamp. Clamp it to the maximum
379    let start_ts = track_ts
380        .checked_add(n_offset_samples)
381        .and_then(|ts| ts.try_into().ok())
382        .unwrap_or_else(|| {
383            warn!("cuesheet track index offset too large, clamping to maximum");
384            i64::MAX
385        });
386
387    Ok(Chapter {
388        start_time: tb.calc_time_saturating(Timestamp::from(start_ts)),
389        end_time: None,
390        start_byte: None,
391        end_byte: None,
392        tags: vec![Tag::new_from_parts(
393            "INDEX",
394            idx_number,
395            Some(StandardTag::CdTrackIndex(idx_number)),
396        )],
397        visuals: Vec::new(),
398    })
399}

401/// Convert a buffer slice containing most ASCII into a string identifier.
402///
403/// For each byte in the buffer, either convert and append it as a character if it is printable
404/// ASCII character, or append it as an escaped hex value.
405fn escape_identifier(buf: &[u8]) -> String {
406    let mut ident = String::with_capacity(buf.len());
407
408    for &byte in buf {
409        if byte.is_ascii() && !byte.is_ascii_control() {
410            ident.push(char::from(byte));
411        }
412        else {
413            let u = (byte & 0xf0) >> 4;
414            let l = byte & 0x0f;
415            ident.push_str("\\0x");
416            ident.push(if u < 10 { (b'0' + u) as char } else { (b'a' + u - 10) as char });
417            ident.push(if l < 10 { (b'0' + l) as char } else { (b'a' + l - 10) as char });
418        }
419    }
420
421    ident
