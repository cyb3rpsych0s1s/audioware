69    /// Parse a RIFF INFO chunk into a `Tag` using the chunk's identifier and a slice containing the
70    /// chunk's payload.
71    pub fn parse_riff_info_chunk(
72        chunk_id: [u8; 4],
73        buf: &[u8],
74        builder: &mut MetadataBuilder,
75    ) -> Result<()> {
76        // It is invalid for a chunk ID to contain non-ASCII characters, or ASCII control characters
77        // per EA-IFF-85. Since chunk IDs are supposed to be well-defined, and each metadata/info
78        // chunk has a standardized ID, this is a hard error.
79        if chunk_id.iter().any(|c| !c.is_ascii() || c.is_ascii_control()) {
80            return decode_error("meta (riff): chunk ID is invalid");
81        }
82
83        // Safety: Key is always ASCII.
84        let key = str::from_utf8(&chunk_id).expect("chunk_id contains only ASCII bytes");
85        let value = String::from_utf8_lossy(buf);
86
87        builder.add_mapped_tags(RawTag::new(key, value), &RIFF_INFO_MAP);
88
89        Ok(())
90    }

93#[cfg(feature = "riff-id3")]
94mod id3 {
95    //! RIFF ID3 chunk metadata format reading.
96
97    use symphonia_core::errors::Result;
98    use symphonia_core::io::ReadBytes;
99    use symphonia_core::meta::{MetadataBuilder, MetadataRevision, MetadataSideData};
100
101    /// Read a RIFF ID3 chunk payload.
102    pub fn read_riff_id3_chunk<B: ReadBytes>(
103        reader: &mut B,
104        side_data: &mut Vec<MetadataSideData>,
105    ) -> Result<MetadataRevision> {
106        let mut builder = MetadataBuilder::new(crate::id3v2::ID3V2_METADATA_INFO);
107        crate::id3v2::read_id3v2(reader, &mut builder, side_data)?;
108        Ok(builder.build())
109    }
110}
