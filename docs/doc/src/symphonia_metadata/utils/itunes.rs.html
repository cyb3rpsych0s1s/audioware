85/// Try to parse iTunes metadata from a raw tag and add it to the builder.
86pub fn parse_itunes_tag(key: String, value: RawValue, builder: &mut MetadataBuilder) -> Result<()> {
87    builder.add_mapped_tags(RawTag::new(key, value), &ITUNES_TAG_MAP);
88    Ok(())
