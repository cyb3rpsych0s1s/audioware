18// A pair of standard tags.
19pub type StandardTagPair = [Option<StandardTag>; 2];
20
21/// A parser of a raw tag value, yielding 0 to 2 standard tags.
22pub type RawTagParser = fn(Arc<String>) -> StandardTagPair;
23/// Maps raw tag key to a raw tag parser.
24pub type RawTagParserMap = HashMap<&'static str, RawTagParser>;
25
26/// Extension trait for `MetadataBuilder` to parsing and adding raw tags.
27pub trait MetadataBuilderExt {
28    /// Attempt to map a raw tag into oneor more standard tags after finding a parser in the
29    /// provided map of raw tag parsers, and then add them to self. If the raw tag cannot be
30    /// mapped, only adds the raw tag to self.
31    fn add_mapped_tags(&mut self, raw: RawTag, parsers: &RawTagParserMap) -> &mut Self;
32
33    /// Attempt to map a raw tag into one or more standard tags using the provided raw tag
34    /// parser, and then add them to self. If the raw tag cannot be mapped, only adds the raw
35    /// tag to self.
36    fn add_mapped_tags_with_parser(&mut self, raw: RawTag, parser: RawTagParser) -> &mut Self;
37}
38
39impl MetadataBuilderExt for MetadataBuilder {
40    fn add_mapped_tags_with_parser(&mut self, raw: RawTag, parser: RawTagParser) -> &mut Self {
41        if let RawValue::String(value) = &raw.value {
42            // Parse and add standard tags.
43            match parser(value.clone()) {
44                [Some(std), None] => {
45                    // One raw tag yielded one standard tag.
46                    return self.add_tag(Tag::new_std(raw, std));
47                }
48                [None, Some(std)] => {
49                    // One raw tag yielded one standard tag.
50                    return self.add_tag(Tag::new_std(raw, std));
51                }
52                [Some(std0), Some(std1)] => {
53                    // One raw tag yielded two standards tags.
54                    self.add_tag(Tag::new_std(raw.clone(), std0));
55                    return self.add_tag(Tag::new_std(raw, std1));
56                }
57                // The raw value could not be parsed.
58                _ => (),
59            }
60        };
61
62        // Could not parse, add a raw tag.
63        self.add_tag(Tag::new(raw))
64    }
65
66    fn add_mapped_tags(&mut self, raw: RawTag, parsers: &RawTagParserMap) -> &mut Self {
67        // Find a parser based on a case insensitive key map.
68        match parsers.get(raw.key.to_ascii_lowercase().as_str()) {
69            Some(parser) => self.add_mapped_tags_with_parser(raw, *parser),
70            None => self.add_tag(Tag::new(raw)),
71        }
72    }
73}

333pub fn parse_itunes_content_advisory(v: Arc<String>) -> StandardTagPair {
334    let content_advisory = v
335        .parse::<u8>()
336        .ok()
337        .and_then(|value| match value {
338            0 => Some(ContentAdvisory::None),
339            1 | 4 => Some(ContentAdvisory::Explicit),
340            2 => Some(ContentAdvisory::Censored),
341            _ => None,
342        })
343        .map(StandardTag::ContentAdvisory);
344
345    [content_advisory, None]
346}

348pub fn parse_id3v2_genre(v: Arc<String>) -> StandardTagPair {
349    use regex_lite::Regex;
350
351    use crate::utils::id3v1::get_genre_name;
352
353    // Regex that will match the following strings:
354    //
355    // "<NUMBER>"
356    // "<NAME>"
357    // "(<NUMBER>)"
358    // "(<NUMBER)<NAME>"
359    let re = Regex::new(r"^(?P<num0>[0-9]+)$|(?:\((?P<num1>[0-9]+)\))?(?P<name>.+)?$")
360        .expect("valid regex");
361
362    // The regex always matches (even an empty string).
363    let caps = re.captures(v.as_str()).expect("regex always matches");
364
365    let name = if let Some(name) = caps.name("name") {
366        // A user-defined genre name provided.
367        Some(name.as_str().to_owned())
368    }
369    else if let Some(num) = caps.name("num0").or_else(|| caps.name("num1")) {
370        // Only genre number provided. Parse to u8, then lookup the genre name.
371        num.as_str().parse::<u8>().ok().and_then(get_genre_name)
372    }
373    else {
374        // Empty string.
375        None
376    };
377
378    // Fallback to the original value for the genre if one could not be parsed.
379    let genre = name
380        .map(|name| StandardTag::Genre(Arc::new(name)))
381        .unwrap_or_else(|| StandardTag::Genre(v));
382
383    [Some(genre), None]
384}

386fn parse_bool(v: Arc<String>) -> Option<bool> {
387    match v.to_ascii_lowercase().as_str() {
388        "1" | "true" | "yes" | "y" => Some(true),
389        "0" | "false" | "no" | "n" => Some(false),
390        _ => None,
391    }
392}

394/// Parse a string in the format "NUM/TOTAL" or "NUM" into a pair of optional integers.
395fn parse_m_of_n(v: Arc<String>) -> (Option<u64>, Option<u64>) {
396    use regex_lite::Regex;
397
398    let re = Regex::new(r"^(?P<m>[0-9]+)(/(?P<n>[0-9]+))?$").expect("valid regex");
399
400    let mut opt_m = None;
401    let mut opt_n = None;
402
403    if let Some(caps) = re.captures(v.as_str()) {
404        opt_m = caps.name("m").and_then(|m| m.as_str().parse::<u64>().ok());
405        opt_n = caps.name("n").and_then(|n| n.as_str().parse::<u64>().ok());
406    }
407
408    (opt_m, opt_n)
409}
